\begin{frame}{Povestea acestui capitol: Problema}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \textit{\textbf{Întrebarea centrală:} Putem identifica regimurile endogene de bulă speculativă înainte ca acestea să se încheie --- sau crahurile sunt cu adevărat aleatorii?}
    \vspace{0.2cm}
    \begin{block}{Problema}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Ipoteza Piețelor Eficiente susține că crahurile sunt șocuri aleatorii \textbf{impredictibile}
            \item Totuși, crahurile prezintă \textbf{tipare recurente}: prețuri în accelerare, comportament de turmă, oscilații
            \item Fizica \textbf{tranzițiilor de fază} oferă o alternativă: crahurile ca evenimente critice endogene
        \end{itemize}
    \end{block}
    \vspace{0.1cm}
    \begin{alertblock}{Dificultatea}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Modelul LPPL are \textbf{7 parametri} cu o suprafață de cost puternic neliniară
            \item Rezultatele sunt \textbf{sensibile la selecția ferestrei} --- când începe bula?
            \item \textbf{Falsele pozitive} și \textbf{supraajustarea} sunt riscuri serioase
        \end{itemize}
    \end{alertblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Povestea acestui capitol: Soluția}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \textit{\textbf{Foaia de parcurs:} De la analogia cu fizica la diagnosticarea regimurilor de bulă speculativă.}
    \vspace{0.2cm}
    \begin{block}{De la teorie la practică}
        \begin{itemize}\setlength{\itemsep}{4pt}
            \item \textbf{Modelul Ising}: explică cum traderii independenți devin o turmă (tranziție de fază)
            \item \textbf{Invarianța de scară discretă}: explică de ce bulele speculative prezintă oscilații log-periodice
            \item \textbf{Ecuația LPPL}: captează creșterea super-exponențială + oscilațiile care se apropie de o tranziție critică de regim
            \item \textcolor{Crimson}{\textbf{Indicatorul de Încredere LPPLS}}: abordare multi-scară pentru monitorizare în timp real
        \end{itemize}
    \end{block}
    \vspace{0.2cm}
    \textit{\textbf{Ideea principală:} LPPL este un ``termometru al febrei'' pentru bule speculative --- citirile ridicate semnalează risc, nu certitudine.}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Ipoteza Piețelor Eficiente}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Afirmațiile fundamentale ale EMH {\scriptsize[\href{https://doi.org/10.2307/2325486}{\textcolor{white}{Fama, 1970}}]}}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Prețurile \textbf{reflectă pe deplin} toată informația disponibilă
                    \item Variațiile de preț sunt \textbf{impredictibile} --- mers aleatoriu
                    \item \textbf{Nicio strategie de tranzacționare} nu poate bate sistematic piața
                    \item Crahurile mari rezultă din \textbf{șocuri externe neprevăzute}
                \end{enumerate}
            \end{block}
            \vspace{-2mm}
            \begin{alertblock}{Implicație pentru managementul riscului}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Momentul crahului nu poate fi prezis din prețuri
                    \item Managementul riscului se bazează pe diversificare, teste de stres și controlul expunerii
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Dovezi în sprijin}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Administratorii profesioniști de fonduri bat rareori indicii [Malkiel, 1973]
                    \item Analiza tehnică are putere predictivă limitată
                    \item Informația este rapid încorporată în prețuri
                \end{itemize}
            \end{exampleblock}
            \vspace{-2mm}
            \begin{block}{Recunoașterea prin Premiul Nobel}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Eugene Fama a primit Premiul Nobel în 2013 pentru EMH
                    \item \textit{``The market is efficient enough that most people can't beat it.''}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Alternativa sistemelor complexe}

    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Piețele ca sisteme complexe {\scriptsize[\href{https://press.princeton.edu/books/paperback/9780691175959/why-stock-markets-crash}{\textcolor{white}{Sornette, 2003}}]}}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Sisteme adaptive complexe} cu comportament emergent
                    \item Supuse \textbf{buclelor de feedback pozitiv} (comportament de turmă)
                    \item Capabile de \textbf{criticalitate auto-organizată}
                    \item Guvernate de \textbf{legi universale} comune cu fizica
                \end{enumerate}
            \end{alertblock}
            \vspace{-1mm}
            \begin{block}{Ideea cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Multe crahuri sunt \textbf{endogene} --- apar din dinamica internă, nu din șocuri externe
                    \item Bula \textit{își creează} propria distrugere
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Dovezi împotriva EMH pure}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Cozi groase:} Evenimente extreme mult mai frecvente decât în distribuția Gaussiană
                    \item \textbf{Gruparea volatilității:} Perioadele calme și turbulente se grupează
                    \item \textbf{Bule speculative:} Prețurile deviază masiv de la fundamentale
                    \item \textbf{Flash crash:} Crahul din 2010 nu a avut un declanșator extern
                \end{itemize}
            \end{exampleblock}
            \vspace{-1mm}
            \begin{block}{Premiul Nobel (în același an!)}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Robert Shiller a câștigat \textit{tot} Premiul Nobel din 2013 pentru demonstrarea faptului că piețele pot fi \textbf{previzibil iraționale}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }

\begin{frame}{Înțelegerea creșterii super-exponențiale}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Creștere exponențială normală}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Evaluarea standard a activelor: $P(t) = P_0 e^{rt}$
                    \item Rata de creștere: $\frac{d\ln P}{dt} = r$ (\textbf{constantă})
                    \item Prețul se dublează la fiecare $T_2 = \ln 2 / r$ perioade
                    \item \textbf{Sustenabilă pe termen nelimitat}
                    \item Descrie randamentele de piață pe termen lung
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Exemplu: S\&P 500}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Randament istoric $\approx$ 10\%/an
                    \item Timp de dublare $\approx$ 7 ani
                    \item Creștere consistentă pe parcursul a peste 100 de ani
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Creștere super-exponențială (de bulă speculativă)}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item În timpul bulelor speculative: $P(t) \sim (t_c - t)^{-\alpha}$, $\alpha > 0$
                    \item Rata de creștere: $\frac{d\ln P}{dt} = \frac{\alpha}{t_c - t}$ (\textbf{în accelerare!})
                    \item Rata $\to \infty$ când $t \to t_c$
                    \item \textbf{Trebuie să se încheie la un moment finit $t_c$}
                    \item Descrie faza de bulă speculativă
                \end{itemize}
            \end{alertblock}

            \begin{block}{Intuiția cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Într-o bulă speculativă, \textit{rata de creștere} continuă să accelereze
                    \item Această accelerare este \textbf{nesustenabilă}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{De ce apare creșterea super-exponențială?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Bucle de feedback pozitiv}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Creșterea prețurilor atrage \textbf{noi cumpărători}
                    \item Noii cumpărători împing prețul \textbf{mai sus}
                    \item Prețurile mai mari atrag \textbf{și mai mulți cumpărători}
                    \item Ciclul se \textbf{accelerează}
                    \item Opusul feedback-ului negativ (revenire la medie) pe care EMH îl presupune
                \end{enumerate}
            \end{block}

            \begin{alertblock}{Mecanisme}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Comportament de turmă:} A urma mulțimea
                    \item \textbf{FOMO:} Frica de a pierde oportunitatea
                    \item \textbf{Efectul de levier:} Prețuri în creștere $\to$ mai mult colateral $\to$ mai multe cumpărări
                    \item \textbf{Media:} Poveștile de succes atrag atenția
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Formulare matematică}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Cererea depinde de randamentele trecute: $D(t) = D_0 + \gamma \cdot \text{Return}_{t-1}$
                    \item Dacă $\gamma > 0$ (urmărirea trendului): $\frac{dP}{dt} \propto D(t) \propto \frac{dP}{dt}\big|_{t-1}$
                    \item Rezultat: $\frac{d^2 \ln P}{dt^2} > 0$ (Super-exponențial!)
                \end{itemize}
            \end{exampleblock}

            \begin{block}{Finalul}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Creșterea super-exponențială \textbf{nu poate continua la infinit}
                    \item La $t_c$: se epuizează noii cumpărători
                    \item Se atinge o constrângere (credit, lichidități)
                    \item Apare un eveniment declanșator
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Modelul Ising: Cel mai simplu comportament colectiv}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Configurarea fizică {\scriptsize[\href{https://doi.org/10.1007/BF02980577}{\textcolor{white}{Ising, 1925}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Fiecare nod $i$ are spin $s_i \in \{-1, +1\}$
                    \item Spinii vecini interacționează cu intensitatea $J$
                    \item Câmpul magnetic extern $h$ poate influența spinii
                    \item Sistemul este în echilibru termic la temperatura $T$
                \end{itemize}
            \end{block}

            \begin{alertblock}{Hamiltonianul}
                $$H = -J \sum_{\langle i,j \rangle} s_i s_j - h \sum_i s_i$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Prima sumă peste perechile de vecini
                    \item $J > 0$: spinii aliniați $=$ energie scăzută
                    \item $h > 0$: spinii în sus $=$ energie scăzută
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Interpretarea financiară {\scriptsize[\href{https://doi.org/10.1017/S1365100500015029}{\textcolor{white}{Cont \& Bouchaud, 2000}}]}}
                \begin{center}
                    \begin{tabular}{ll}
                        \toprule
                        \textbf{Fizică} & \textbf{Finanțe} \\
                        \midrule
                        Spin în sus $+1$ & Ordin de cumpărare \\
                        Spin în jos $-1$ & Ordin de vânzare \\
                        Cuplaj $J$ & Intensitatea comportamentului de turmă \\
                        Câmp $h$ & Semnal fundamental \\
                        Temperatura $T$ & Zgomot / incertitudine \\
                        Magnetizare $m$ & Sentimentul pieței \\
                        \bottomrule
                    \end{tabular}
                \end{center}
            \end{exampleblock}

            \begin{block}{}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $J$ mare (comportament de turmă puternic) + $T$ mic (zgomot redus)

\begin{frame}{Ce sunt oscilațiile log-periodice?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Tiparul}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Se \textbf{comprimă} în timp real (mai rapid lângă $t_c$)
                    \item Sunt \textbf{uniform distanțate} în $\ln(t_c - t)$
                    \item Modulează tendința de tip lege de putere
                    \item Reflectă structura ierarhică a pieței
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Forma matematică}
                $$\ln P(t) = A + B(t_c-t)^m \cdot [1 + C\cos(\omega\ln(t_c-t))]$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Cosinusul creează oscilații în $\ln(t_c - t)$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{De ce ``log-periodice''?}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Reprezentare în funcție de $\tau = \ln(t_c - t)$: $\ln P = A + Be^{m\tau}[1 + C\cos(\omega\tau)]$
                    \item Oscilațiile au \textbf{perioadă constantă} în timp logaritmic
                    \item Perioada $= 2\pi/\omega$, raportul de scalare $\lambda = e^{2\pi/\omega}$
                \end{itemize}
            \end{alertblock}

            \begin{block}{Semnificația fizică}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Fiecare oscilație = o ``rundă'' a bulei speculative
                    \item Prețurile accelerează, se corectează scurt, apoi accelerează și mai repede
                    \item Fiecare rundă este de $\lambda$ ori mai scurtă decât precedenta
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Invarianță continuă de scară}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Definiție}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $f(x)$ este invariantă la scară dacă: $f(\lambda x) = \lambda^{-\alpha} f(x) \;\forall \lambda > 0$
                    \item Scalarea cu \textbf{orice} factor $\lambda$ doar rescalează funcția
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Soluție unică}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Singurele} soluții sunt: $f(x) = C \cdot x^{-\alpha}$
                    \item Legi de putere pure!
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Exemple în finanțe}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Cozile randamentelor:} $P(|r| > x) \sim x^{-\alpha}$
                    \item \textbf{Dimensiunea firmelor:} Legea lui Zipf
                    \item \textbf{Volumele tranzacționate:} Distribuție de tip lege de putere
                    \item \textbf{Impactul asupra prețului:} $\Delta P \sim V^{1/2}$
                    \item Apar atunci când nu există o scară caracteristică
                \end{itemize}
            \end{exampleblock}

            \begin{block}{Limitare}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item CSI pură produce doar legi de putere
                    \item De unde provin oscilațiile?
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Invarianță discretă de scară}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Definiție {\scriptsize[\href{https://doi.org/10.1016/S0370-1573(97)00076-8}{\textcolor{white}{Sornette, 1998}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Invarianță doar pentru $\lambda_0$ \textbf{specific}: $f(\lambda_0 x) = \lambda_0^{-\alpha} f(x)$
                    \item Sistemul are un raport de scară \textbf{preferat}
                \end{itemize}
            \end{block}

            \begin{block}{Soluția generală}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item $f(x) = x^{-\alpha} \cdot G(\ln x)$ unde $G$ este periodică cu perioada $\ln\lambda_0$
                    \item Fourier: $f(x) = x^{-\alpha}\sum_n c_n e^{in\omega\ln x}$, $\omega = 2\pi/\ln\lambda_0$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Aproximare de ordin dominant}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Se păstrează doar $n = 0, \pm 1$: $f(x) \approx x^{-\alpha}[A' + B'\cos(\omega\ln x + \phi)]$
                    \item \textbf{Lege de putere + oscilații log-periodice!}
                \end{itemize}
            \end{exampleblock}

            \begin{block}{Interpretare fizică}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item DSI apare din \textbf{structura ierarhică}
                    \item Geometrie fractală, rețele ierarhice
                    \item Organizare multi-scară
                    \item Fiecare nivel este legat prin factorul $\lambda_0$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{De ce $\lambda \approx 2$?}
    \begin{cminipage}{0.95\textwidth}

    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Observație empirică}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Pe parcursul multor bule speculative: $\lambda = e^{2\pi/\omega} \approx 2 - 2.5$
                    \item Corespunde la $\omega \approx 6 - 8$
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Explicații teoretice}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item \textbf{Ierarhie binară:} Fiecare nivel acoperă de 2x nivelul anterior
                    \item \textbf{Structura rețelei:} Raport de ramificare $\approx 2$
                    \item \textbf{Renormalizare:} Punct fix al fluxului RG
                    \item \textbf{Comportament uman:} Dublare caracteristică
                \end{enumerate}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Implicații}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $\lambda$ universal sugerează o origine structurală profundă
                    \item Nu este specific unei piețe; poate fi folosit ca filtru
                    \item Susține validitatea LPPL
                    \item Dacă $\lambda$ estimat este departe de 2, ajustarea poate fi falsă
                \end{itemize}
            \end{alertblock}

            \begin{block}{Condiție de filtrare}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Se acceptă ajustări doar dacă: $4 \leq \omega \leq 25$
                    \item Echivalent: $1.3 \leq \lambda \leq 4.8$
                    \item Valoare centrală: $\lambda \approx 2$, $\omega \approx 6.3$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Ecuația LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{center}
        \begin{minipage}{0.85\textwidth}
            \begin{block}{}
                \large
                $$\boxed{\ln P(t) = A + B(t_c - t)^m + C(t_c - t)^m \cos(\omega \ln(t_c - t) - \phi)}$$
            \end{block}
        \end{minipage}
    \end{center}

    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Parametri liniari (OLS)}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $A$: Log-prețul la timpul critic $t_c$
                    \item $B$: Amplitudinea legii de putere ($B < 0$)
                    \item $C$: Amplitudinea oscilațiilor
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Parametri neliniari (DE)}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_c$: Timpul critic (fereastra de crah)
                    \item $m$: Exponentul legii de putere ($0 < m < 1$)
                    \item $\omega$: Frecvența log-periodică
                    \item $\phi$: Faza oscilațiilor
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Două parametrizări LPPL echivalente}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Forma cu fază (7 parametri)}
                \begin{equation*}
                    \ln P(t) = A + B\tau^m + C\tau^m \cos(\omega \ln\tau - \phi)
                \end{equation*}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $\tau = t_c - t$; parametri: $A, B, C, t_c, m, \omega, \phi$
                    \item Interpretabil fizic: $C$ = amplitudinea oscilației, $\phi$ = faza
                    \item Utilizat în derivările teoretice
                    \item 4 parametri neliniari: $t_c, m, \omega, \phi$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Forma linearizată (7 parametri)}
                \begin{equation*}
                    \ln P(t) = A + Bf + C_1 g + C_2 h
                \end{equation*}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $f = \tau^m$, $g = \tau^m\cos(\omega\ln\tau)$, $h = \tau^m\sin(\omega\ln\tau)$
                    \item Parametri: $A, B, C_1, C_2, t_c, m, \omega$
                    \item Doar 3 neliniari: $t_c, m, \omega$
                    \item $A, B, C_1, C_2$ rezolvați prin OLS
                    \item Recuperare: $C = \sqrt{C_1^2 + C_2^2}$, $\phi = \operatorname{atan2}(C_2, C_1)$
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}

    \vspace{-1mm}
    \begin{exampleblock}{De ce două forme?}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item Forma linearizată: 4 $\to$ 3 parametri neliniari, convergență mult mai bună
            \item Ambele forme sunt matematic identice --- ajustează aceeași curbă
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 1 al derivării: Teoria bulelor speculative raționale}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Condiția de non-arbitraj {\scriptsize[\href{https://doi.org/10.3386/w0945}{\textcolor{white}{Blanchard \& Watson, 1982}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Bulă speculativă rațională: $\E_t[dP] = P r\, dt + P h(t) \kappa\, dt$
                    \item $r$: Rata fără risc
                    \item $h(t)$: Rata de hazard a crahului (probabilitate/timp)
                    \item $\kappa$: Dimensiunea așteptată a crahului (ex., 30\%)
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Interpretare}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Risc mai mare de crah $\Rightarrow$ randament așteptat mai mare necesar
                    \item Prețul trebuie să crească mai rapid pentru a compensa
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Observație cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Dacă $h(t)$ \textbf{crește}, randamentele așteptate \textbf{accelerează}
                    \item $\frac{d\ln P}{dt} = r + h(t)\kappa$; pe măsură ce $h(t) \uparrow$, creșterea $\uparrow$
                    \item Aceasta creează creștere super-exponențială!
                \end{itemize}
            \end{alertblock}

            \begin{block}{Întrebarea}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Ce formă funcțională ar trebui să aibă $h(t)$?
                    \item Fizica oferă răspunsul prin fenomenele critice!
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 2 al derivării: Ecuația de preț compensată pentru crah}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Configurație}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Prețul $P(t)$ poate suferi un crah la momentul aleatoriu $t^*$
                    \item La crah: $P \to P(1 - \kappa)$, cu $\kappa \in (0,1)$
                    \item Rata de hazard: $h(t) = \Pr[\text{crah în } (t, t+dt)] / dt$
                    \item Înainte de crah, prețul urmează:
                \end{itemize}
                \vspace{1mm}
                $$\frac{dP}{P} = \mu(t)\,dt + \sigma\,dW - \kappa\,dj$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $dj$: proces de salt cu intensitatea $h(t)$
                    \item $\E[dj] = h(t)\,dt$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Condiția de non-arbitraj}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Randamentul așteptat trebuie să egaleze $r$ (rata fără risc):
                \end{itemize}
                $$\E\!\left[\frac{dP}{P}\right] = \mu(t)\,dt - \kappa\,h(t)\,dt = r\,dt$$
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Rezolvând pentru $\mu(t)$:
                \end{itemize}
                $$\boxed{\mu(t) = r + \kappa\, h(t)}$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Risc mai mare de crah $\Rightarrow$ prețul trebuie să crească \textbf{mai rapid}
                    \item Investitorii cer compensație pentru asumarea riscului de crah
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 3 al derivării: De la tendință la log-preț}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{block}{Integrarea ecuației de preț (înainte de crah)}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item Se iau așteptările și se ignoră zgomotul ($\sigma\,dW$) pentru traiectoria medie condiționată:
            \vspace{-2mm}
                $$\frac{d\,\E[\ln P]}{dt} = \mu(t) = r + \kappa\, h(t)$$
            \vspace{-4mm}
            \item Se integrează de la $0$ la $t$:
            \vspace{-2mm}
                $$\E[\ln P(t)] = \ln P(0) + r\,t + \kappa \int_0^t h(s)\,ds$$
            \vspace{-4mm}
            \item Forma $\ln P(t)$ depinde de \textbf{hazardul cumulat} $\int_0^t h(s)\,ds$
            \item Dacă $h(t) = $ constant $\Rightarrow$ creștere exponențială (EMH)
            \item Dacă $h(t)$ \textbf{crește} $\Rightarrow$ \textbf{creștere super-exponențială} (bulă speculativă!)
        \end{itemize}
    \end{block}

    \vspace{-2mm}

    \begin{alertblock}{Întrebarea cheie}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item Ce formă funcțională ar trebui să aibă $h(t)$ în timpul unei bule speculative?
            \item \textbf{Fenomenele critice} (Părțile II--III): în apropierea unui punct critic, mărimile urmează \textbf{legi de putere} cu \textbf{corecții log-periodice}
        \end{itemize}
    \end{alertblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 4 al derivării: De ce hazard sub formă de lege de putere?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{De la fenomene critice la finanțe}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item În apropierea unei tranziții de fază la $T_c$, mărimile fizice diverg ca legi de putere
                    \item Susceptibilitate: $\chi \sim |T - T_c|^{-\gamma}$
                    \item Lungimea de corelație: $\xi \sim |T - T_c|^{-\nu}$
                    \item \textbf{Analogia cu piața:} se înlocuiește $T \to t$, $T_c \to t_c$
                    \item ``Susceptibilitatea'' la crah urmează aceeași scalare critică
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Forma hazardului de tip lege de putere}
                \vspace{-2mm}
                $$\boxed{h(t) = \alpha(t_c - t)^{m-1}, \quad 0 < m < 1}$$
                \vspace{-3mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Cea mai simplă funcție consistentă cu scalarea critică
                    \item $\alpha > 0$: amplitudinea singularității
                    \item $m$ este \textbf{exponentul critic} al bulei speculative
                    \item Analogă claselor de universalitate din fizică
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-2mm}
    \begin{exampleblock}{De ce $0 < m < 1$? Două constrângeri esențiale}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $m < 1 \;\Rightarrow\; m{-}1 < 0 \;\Rightarrow\; h(t) \to \infty$ când $t \to t_c$: crahul devine din ce în ce mai probabil
            \item $m > 0 \;\Rightarrow\; \int_0^{t_c}\! h(s)\,ds < \infty$: hazardul cumulat rămâne finit, deci $\ln P$ rămâne mărginit
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 5 al derivării: Integrarea hazardului de tip lege de putere}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Pregătirea integralei}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Din Pasul 3: $\ln P(t) = \ln P(0) + rt + \kappa\!\int_0^t\! h(s)\,ds$
                    \item Se substituie $h(s) = \alpha(t_c - s)^{m-1}$:
                \end{itemize}
                \vspace{-2mm}
                $$\kappa\!\int_0^t\! h(s)\,ds = \kappa\alpha\!\int_0^t\! (t_c - s)^{m-1}\,ds$$
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Substituție: $u = t_c - s$, $du = -ds$
                    \item Limite: $s{=}0 \!\to\! u{=}t_c$;\; $s{=}t \!\to\! u{=}t_c{-}t$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Calculul integralei}
                \vspace{-2mm}
                \begin{align*}
                    \kappa\alpha\!\int_0^t\! (t_c{-}s)^{m-1}ds &= \kappa\alpha\!\int_{t_c-t}^{t_c}\! u^{m-1}\,du \\[-2pt]
                    &= \frac{\kappa\alpha}{m}\Bigl[t_c^m - (t_c{-}t)^m\Bigr]
                \end{align*}
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Convergentă deoarece $m > 0$
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-2mm}
    \begin{exampleblock}{Gruparea termenilor}
        \vspace{-2mm}
        $$\ln P(t) = \underbrace{\ln P(0) + rt + \tfrac{\kappa\alpha}{m}t_c^m}_{\displaystyle A} + \underbrace{\bigl(-\tfrac{\kappa\alpha}{m}\bigr)}_{\displaystyle B}(t_c - t)^m \qquad\Longrightarrow\qquad \boxed{\ln P(t) = A + B(t_c - t)^m}$$
        \vspace{-3mm}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $B = -\kappa\alpha/m < 0$ deoarece $\kappa, \alpha, m > 0$ --- asigură creștere \textbf{super-exponențială} spre $t_c$
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 6 al derivării: Integrala oscilatorie log-periodică}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{DSI modifică rata de hazard}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Din DSI (Partea III), hazardul capătă oscilații:
                \end{itemize}
                \vspace{-2mm}
                $$h(t) = \alpha(t_c{-}t)^{m-1}\bigl[1 + c\cos(\omega\ln(t_c{-}t) {-} \phi')\bigr]$$
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Pasul 5 a tratat termenul ``$1$'' $\to$ $B(t_c{-}t)^m$
                    \item Acum se evaluează corecția oscilatorie:
                \end{itemize}
                \vspace{-2mm}
                $$I = \int u^{m-1}\cos(\omega\ln u)\,du$$
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Trucul exponențialei complexe}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Se scrie $\cos(\omega\ln u) = \operatorname{Re}[u^{i\omega}]$
                    \item Deci $I = \operatorname{Re}\!\left[\int u^{m-1+i\omega}\,du\right] = \operatorname{Re}\!\left[\frac{u^{m+i\omega}}{m+i\omega}\right]$
                    \item Forma polară: $m + i\omega = \sqrt{m^2+\omega^2}\;e^{i\theta}$
                    \item unde $\theta = \arctan(\omega/m)$
                \end{itemize}
                \vspace{-2mm}
                $$\boxed{I = \frac{u^m}{\sqrt{m^2+\omega^2}}\cos(\omega\ln u - \theta)}$$
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-1mm}
    \begin{exampleblock}{Combinarea termenilor de lege de putere + oscilatorii $\longrightarrow$ Ecuația LPPL}
        \vspace{-1mm}
        $$\ln P(t) = A + B(t_c{-}t)^m + C(t_c{-}t)^m\cos\bigl(\omega\ln(t_c{-}t) - \phi\bigr)$$
        \vspace{-3mm}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $C = -\kappa\alpha c\,/\,\sqrt{m^2 + \omega^2}$; \quad $\phi = \phi' + \arctan(\omega/m)$
            \item Defazajul $\phi$ apare natural din integrare --- \textbf{nu} este un parametru liber
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Pasul 7 al derivării: Lanțul complet}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{block}{De la așteptări raționale la LPPL --- Rezumat {\scriptsize[\href{https://doi.org/10.48550/arXiv.cond-mat/9903321}{\textcolor{white}{Johansen et al., 2000}}]}}
        \begin{enumerate}\setlength{\itemsep}{0pt}
            \item \textbf{Non-arbitraj:} $\E[dP/P] = r\,dt$ $\Longrightarrow$ $\mu(t) = r + \kappa\,h(t)$
            \item \textbf{Integrare:} $\E[\ln P(t)] = \ln P(0) + r\,t + \kappa\!\int_0^t h(s)\,ds$
            \item \textbf{Fenomene critice:} $h(t) = \alpha(t_c - t)^{m-1}$ cu $0 < m < 1$
            \item \textbf{Integrala legii de putere:} $\kappa\!\int_0^t h(s)\,ds = \tfrac{\kappa\alpha}{m}t_c^m - \tfrac{\kappa\alpha}{m}(t_c - t)^m$ $\;\Rightarrow\;$ $\ln P = A + B(t_c-t)^m$
            \item \textbf{Corecția DSI:} $h(t) \to \alpha(t_c{-}t)^{m-1}[1 + c\cos(\omega\ln(t_c{-}t) {-} \phi')]$
            \item \textbf{Integrala oscilatorie:} $\int u^{m-1}\cos(\omega\ln u)\,du = \frac{u^m}{\sqrt{m^2+\omega^2}}\cos(\omega\ln u - \theta)$
        \end{enumerate}
    \end{block}
    \vspace{-2mm}
    \begin{exampleblock}{}
        \vspace{-1mm}
        \begin{center}
            \large $\boxed{\ln P(t) = A + B(t_c - t)^m + C(t_c - t)^m \cos\bigl(\omega \ln(t_c - t) - \phi\bigr)}$
        \end{center}
        \vspace{-2mm}
    \end{exampleblock}
    \vspace{-2mm}
    \begin{alertblock}{Constrângeri cheie din derivare}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $B = -\kappa\alpha/m < 0$ (creștere super-exponențială); \; $0 < m < 1$ (singularitate + preț finit)
            \item $C = -\kappa\alpha c/\sqrt{m^2{+}\omega^2}$; \; $\phi = \phi' + \arctan(\omega/m)$ (defazaj din integrare)
        \end{itemize}
    \end{alertblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Interpretarea parametrilor LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Timpul critic $t_c$}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Cel mai probabil moment pentru schimbarea de regim
                    \item \textbf{Nu} este data exactă a crahului
                    \item Sfârșitul bulei speculative, începutul unui nou regim
                    \item Incertitudine: de regulă $\pm$ săptămâni până la luni
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Exponentul legii de putere $m$}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Controlează rata de accelerare
                    \item $m < 1$: Singularitate în timp finit
                    \item Empiric: $m \approx 0.3 - 0.8$
                    \item $m$ mai mic = accelerare mai rapidă
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Amplitudine $B < 0$}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Trebuie să fie negativ pentru bulă speculativă
                    \item Magnitudinea este legată de intensitatea bulei speculative
                    \item $B > 0$ ar fi anti-bulă
                \end{itemize}
            \end{alertblock}

            \begin{block}{Frecvența unghiulară $\omega$}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Numărul de oscilații $\approx \omega/(2\pi)$
                    \item Legată de ierarhie: $\lambda = e^{2\pi/\omega}$
                    \item Empiric: $\omega \approx 4 - 25$
                    \item Universal: $\lambda \approx 2$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Studiu de caz Bitcoin: rezultate LPPL cu intervale de încredere}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Date}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Activ:} Bitcoin (BTC-USD)
                    \item \textbf{Fereastră:} 20 iul -- 8 nov, 2021
                    \item \textbf{Observații:} 111 prețuri zilnice
                    \item \textbf{Interval de preț:} \$29,000 $\to$ \$69,000
                \end{itemize}
            \end{block}

            \begin{block}{Metoda de estimare}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Linearizare parțială (OLS pentru $A, B, C$)
                    \item Differential Evolution pentru $(t_c, m, \omega, \phi)$
                    \item \textbf{IC Bootstrap:} 100 de reeșantionări ale reziduurilor
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Parametri estimați cu IC 95\%}
                \begin{center}

\begin{frame}{De ce este dificilă estimarea LPPL?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Provocări}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item \textbf{Neliniaritate:} 4 parametri neliniari
                    \item \textbf{Minime multiple:} Suprafața de cost este accidentată
                    \item \textbf{Corelarea parametrilor:} $t_c$ și $m$ interacționează
                    \item \textbf{Cvasi-periodicitate:} $\omega$ creează minime locale
                    \item \textbf{Sensibilitate:} Modificări mici $\to$ efecte mari
                \end{enumerate}
            \end{alertblock}

            \begin{block}{Abordarea naivă eșuează}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Rămâne blocată în minime locale
                    \item Sensibilă la estimarea inițială
                    \item Poate converge la valori nefizice
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Soluție: Abordare în două etape}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item \textbf{Linearizare parțială:} Se fixează $(t_c, m, \omega, \phi)$, se rezolvă $(A, B, C)$ prin OLS (7D $\to$ 4D)
                    \item \textbf{Optimizare globală:} Evoluție diferențială pentru $(t_c, m, \omega, \phi)$ --- căutare bazată pe populații, regiuni multiple
                \end{enumerate}
            \end{exampleblock}

            \begin{block}{Ideea cheie {\scriptsize[\href{https://doi.org/10.1016/j.physa.2013.04.012}{\textcolor{white}{Filimonov \& Sornette, 2013}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Subproblema OLS este rapidă și exactă
                    \item Optimizarea exterioară necesită doar 4 parametri
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Metoda linearizării parțiale}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Rescrierea LPPL (Forma linearizată)}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Baza: $f_1 = 1$, $f_2 = (t_c - t)^m$, $f_3 = f_2\cos(\omega\ln(t_c-t))$, $f_4 = f_2\sin(\omega\ln(t_c-t))$
                    \item Atunci: $\ln P(t) = Af_1 + Bf_2 + C_1 f_3 + C_2 f_4$
                    \item Aceasta este \textbf{liniară} în $(A, B, C_1, C_2)$!
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Soluția OLS}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Dat fiind $(t_c, m, \omega)$:
                    \item $(A, B, C_1, C_2)' = (F'F)^{-1}F'\mathbf{y}$ unde $\mathbf{y} = \ln P$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Optimizarea exterioară}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Minimizare $\text{SSR} = \sum_t [\ln P(t) - \widehat{\ln P}(t)]^2$
                    \item $\widehat{\ln P}$ folosește $(A, B, C_1, C_2)$ optimi OLS
                    \item Se utilizează \textbf{Evoluția diferențială}:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Populație de soluții candidat
                        \item Operatori de mutație și încrucișare
                        \item Converge la optimul global
                    \end{itemize}
                \end{itemize}
            \end{alertblock}

            \begin{block}{Limite}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_c \in [T, T + 0.5(T - t_1)]$
                    \item $m \in [0.1, 0.9]$, $\omega \in [4, 25]$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Implementare: Algoritmul de ajustare LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Descrierea algoritmului}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Intrare:} Indexul temporal $t$, log-prețuri $\ln P$
                    \item \textbf{Buclă exterioară:} DE optimizează $(t_c, m, \omega)$
                    \item \textbf{Pas interior:} OLS rezolvă $(A, B, C_1, C_2)$ prin matricea de bază $\mathbf{F}$
                    \item \textbf{Cost:} $\sum_i (\ln P_i - \hat{P}_i)^2$
                    \item \textbf{Ieșire:} 7 parametri optimi
                \end{enumerate}
            \end{block}
            \vspace{-1mm}
            \begin{exampleblock}{Matricea de bază $\mathbf{F}$}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $f_{i1} = 1$ (interceptul)
                    \item $f_{i2} = (t_c - t_i)^m$ (legea puterii)
                    \item $f_{i3} = f_{i2}\cos(\omega\ln(t_c - t_i))$
                    \item $f_{i4} = f_{i2}\sin(\omega\ln(t_c - t_i))$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Setări DE}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Populație: $30 \times$ dimensiuni
                    \item Mutație: $(0.5, 1.5)$, Încrucișare: 0.7
                    \item Iterații maxime: 2000, Tol: $10^{-12}$
                    \item Semințe multiple pentru robustețe
                \end{itemize}
            \end{alertblock}
            \vspace{-1mm}
            \begin{block}{Biblioteci utilizate}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \href{https://numpy.org}{\texttt{numpy}} --- algebră liniară, matricea de bază
                    \item \href{https://docs.scipy.org/doc/scipy/reference/generated/scipy.optimize.differential\_evolution.html}{\texttt{scipy.optimize}} --- evoluție diferențială
                    \item \href{https://numpy.org/doc/stable/reference/generated/numpy.linalg.lstsq.html}{\texttt{numpy.linalg.lstsq}} --- subproblema OLS
                    \item \href{https://pypi.org/project/yfinance/}{\texttt{yfinance}} --- descărcarea datelor
                \end{itemize}
            \end{block}

                \begin{itemize}\setlength{\itemsep}{0pt}

        \begin{column}{0.47\textwidth}

                    \item Varierea sistematică a lui $t_1$
                    \item Menținerea lui $t_2$ fix (astăzi)
                    \item Dimensiuni ale ferestrei: 60--750 zile
                    \item Agregarea rezultatelor
                \end{enumerate}

\begin{frame}{Previzualizare: De la ajustare la validare}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{1. Diagnosticul reziduurilor}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Ljung-Box}: $Q = n(n+2)\sum_{k=1}^{h}\frac{\hat{\rho}_k^2}{n-k}$
                    \item \textbf{Testul ARCH-LM}: Fără heteroscedasticitate
                    \item \textbf{Jarque-Bera}: Normalitatea reziduurilor
                \end{itemize}
            \end{block}

            \begin{alertblock}{2. Testul raportului de verosimilitate}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $LR = 2(\ell_{LPPL} - \ell_{null}) \sim \chi^2(df = 4)$
                    \item $H_0$: Modelul exponențial este suficient
                    \item $H_1$: LPPL oferă o ajustare mai bună
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{3. Intervale de încredere bootstrap}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Re-eșantionarea reziduurilor (bootstrap pe blocuri)
                    \item Re-estimarea LPPL de peste 100 de ori
                    \item Calculul IC 95\% pentru $t_c$, $m$, $\omega$
                \end{enumerate}
            \end{exampleblock}

            \begin{block}{4. Validare în afara eșantionului}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Ajustare pe date până la $t^*$, estimarea $t_c$ și a traiectoriei prețului
                    \item Comparare cu valorile realizate
                    \item Urmărirea erorii de estimare pe ferestre rulante
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Modele de referință: Ce trebuie să depășească LPPL?}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Referințe de trend}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item \textbf{Trend exponențial:} $\ln P = a + bt$
                    \item \textbf{Log-preț cuadratic:} $\ln P = a + bt + ct^2$
                    \item \textbf{Semnal de impuls:} $P_t / P_{t-k} - 1 > \theta$
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Referințe de risc}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item \textbf{Depășirea volatilității:} $\sigma_t > k \cdot \bar{\sigma}$
                    \item \textbf{Regulă simplă de drawdown:} ieșire după $-x\%$
                    \item \textbf{Placebo LPPL cu fereastră aleatorie:} ajustarea LPPL pe ferestre amestecate sau cu start aleatoriu
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Metrici de evaluare}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{AIC / BIC:} parsimonia modelului
                    \item \textbf{RMSE / MAE în afara eșantionului}
                    \item \textbf{Acuratețe direcțională:} semnal înainte de ruptura de regim?
                    \item \textbf{Rata de fals pozitivi:} alarme fără crah
                    \item \textbf{Timp de avertizare} și \textbf{drawdown evitat}
                    \item \textbf{Stabilitatea semnalului} pe ferestre rulante
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Mesajul cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Acceptați LPPL doar dacă adaugă \textbf{informație diagnostică incrementală} față de regulile mai simple de trend, impuls și volatilitate
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Baterie de validare statistică (1/2)}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Teste de screening pre-LPPL}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Trend exponențial:} $\ln P_t = a + bt + \varepsilon_t$
                    \item \textbf{Convexitate cuadratică:} $\ln P_t = a + bt + ct^2 + \varepsilon_t$; necesită $c > 0$
                    \item \textbf{Lege a puterii vs.\ exponențială:} AIC/BIC
                    \item \textbf{Ruptură structurală:} Testul Chow / sup-Wald
                \end{itemize}
            \end{block}
            \vspace{-2mm}
            \begin{alertblock}{Teste de comparare a modelelor}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{AIC / BIC:} exponențial vs.\ cuadratic vs.\ LPPL
                    \item \textbf{Testul Vuong} pentru comparație non-imbricată
                    \item \textbf{RMSE / MAE cu validare încrucișată}
                    \item \textbf{Diebold--Mariano} pentru comparația erorilor de prognoză
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Diagnosticul reziduurilor}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Ljung--Box:} autocorelația reziduurilor
                    \item \textbf{ARCH-LM:} heteroscedasticitate condițională
                    \item \textbf{Jarque--Bera:} non-normalitate
                    \item \textbf{Breusch--Pagan / White:} heteroscedasticitate
                    \item \textbf{Testul secvențelor:} modele ale semnelor reziduurilor
                    \item \textbf{Testul BDS:} dependență neliniară
                \end{itemize}
            \end{exampleblock}
            \vspace{-2mm}
            \begin{block}{Principiu cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Evidența LPPL necesită o \textbf{convergență a testelor}, nu un singur $R^2$ ridicat
                    \item Credibilă doar atunci când accelerarea, validitatea parametrilor, diagnosticul reziduurilor și comparația cu referințele \textbf{toate} sunt de acord
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Baterie de validare statistică (2/2)}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Teste de robustețe a semnalului de bulă speculativă}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Rata de trecere multi-fereastră:} fracțiunea ferestrelor care trec toate filtrele
                    \item \textbf{Gruparea $t_c$:} ferestrele diferite sunt de acord asupra $t_c$?
                    \item \textbf{Lățimea IC bootstrap:} IC îngust $\Rightarrow$ semnal stabil
                    \item \textbf{Placebo cu fereastră aleatorie:} LPPL pe sub-perioade aleatorii; numărarea fals pozitivilor
                    \item \textbf{Placebo cu randamente amestecate:} permutarea randamentelor, re-ajustare; LPPL ar trebui să eșueze
                    \item \textbf{Control negativ:} aplicare pe perioade fără bulă speculativă
                    \item \textbf{Backtest rulant:} urmărirea LPPLS CI ca în timp real
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{De ce contează testele placebo}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL are 7 parametri $\Rightarrow$ poate ajusta zgomotul dacă nu este constrâns
                    \item Un model care ajustează și date aleatorii nu oferă \textbf{nicio evidență}
                    \item Rata de trecere a placebo-ului ar trebui să fie scăzută și raportată explicit
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{exampleblock}{Standard de raportare}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item De raportat: rata de adevărat pozitiv, rata de fals pozitiv, timpul de avertizare, distribuția erorii $t_c$
                    \item Un semnal care supraviețuiește doar selecției ex-post nu constituie evidență
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Intervale de încredere bootstrap: Algoritm}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Algoritmul bootstrap pe blocuri}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Ajustarea LPPL pe datele originale $\Rightarrow \hat{\theta}$
                    \item Calculul reziduurilor: $e_i = \ln P_i - \widehat{\ln P}_i$
                    \item Dimensiunea blocului $b = \lfloor\sqrt{N}\rfloor$
                    \item Pentru $k = 1, \ldots, B$ (ex., $B = 500$):
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Re-eșantionarea blocurilor de reziduuri cu înlocuire
                        \item Crearea prețurilor sintetice: $P^*_i = \exp(\widehat{\ln P}_i + e^*_i)$
                        \item Re-ajustarea LPPL $\Rightarrow \hat{\theta}^{(k)}$
                        \item Stocarea $t_c^{(k)}$ dacă filtrele trec
                    \end{itemize}
                    \item IC: $[t_c^{(0.025)},\; t_c^{(0.975)}]$
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{De ce bootstrap pe blocuri?}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Reziduurile LPPL sunt \textbf{autocorelate}
                    \item Bootstrap-ul simplu distruge dependența serială
                    \item Bootstrap-ul pe blocuri păstrează structura locală
                    \item Dimensiunea blocului $b \approx \sqrt{N}$ este standard
                \end{itemize}
            \end{alertblock}

            \begin{exampleblock}{Verificări de calitate}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Rata bootstrap validă $> 70\%$
                    \item Distribuția $t_c$ este unimodală
                    \item Lățimea IC $< 0.4T$ (nu prea larg)
                    \item Diagnostic ex-post: vârful real în cadrul IC 95\%
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
    }
    \hfill\quantlet{TSA\_ch13\_bootstrap\_ci}{https://github.com/QuantLet/TSA/tree/main/TSA\_ch13/TSA\_ch13\_bootstrap\_ci}
    \end{cminipage}
\end{frame}

\begin{frame}{Cum știm că intrăm într-o bulă speculativă?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Indicatori la nivel de piață}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Convexitate în log-prețuri (randamente în accelerare)
                    \item Rată de creștere super-exponențială
                    \item Feedback pozitiv și comportament de turmă
                    \item Narațiuni extreme, FOMO, frenezie mediatică
                    \item Expansiunea levierului și creșterea lichidității
                \end{enumerate}
            \end{block}

            \begin{exampleblock}{Diagnostice specifice LPPL}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \setcounter{enumi}{5}
                    \item Toate cele 8 condiții de filtrare trec
                    \item Robustețe multi-fereastră a ajustărilor
                    \item Gruparea valorilor $t_c$ estimate
                    \item Intervale de încredere bootstrap strânse
                    \item Confirmare din indicatori fundamentali
                \end{enumerate}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Principiu cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Nu observăm direct o bulă speculativă --- o \textbf{diagnosticăm} din evidențe convergente
                    \item Niciun indicator individual nu este suficient
                    \item Forța provine din \textbf{convergența} semnalelor multiple
                \end{itemize}
            \end{alertblock}

            \begin{block}{Mesajul central}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Un semnal de bulă speculativă este \textbf{probabilistic}
                    \item Fragilitate și risc de crah crescute, nu certitudine
                    \item Răspunsul adecvat: gestionarea riscului, nu panică
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Standard științific pentru evidența LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Ce trebuie raportat}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Modul de analiză:} timp real, pseudo-RT sau ex-post
                    \item Universul de active, sursa datelor, datele de start/final
                    \item Dacă grila de ferestre a fost \textbf{pre-specificată} sau post-hoc
                    \item Limitele parametrilor și setările optimizatorului
                    \item Condițiile de filtrare utilizate (toate 8)
                    \item Metoda bootstrap și $N$; modelele de referință testate
                    \item \textbf{Toate} semnalele eșuate și reușite
                    \item Eroarea $t_c$ și rata de fals pozitivi
                    \item Dacă procedura a fost aplicată pe \textbf{perioade fără bulă speculativă}
                    \item \textbf{Reproductibilitatea} codului și datelor
                \end{enumerate}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{Ce contează ca o confirmare?}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Vârful în cadrul IC 95\% pentru $t_c$
                    \item Drawdown $> 20\%$ în 3 luni de la $t_c$
                    \item LPPLS CI crescut înainte de vârf
                    \item Rezultatul supraviețuiește testelor de referință și placebo
                \end{itemize}
            \end{block}
            \vspace{-2mm}
            \begin{alertblock}{Avertizare privind data snooping}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Studiile LPPL sunt vulnerabile la data snooping: pot fi căutate multe active, ferestre și limite ale parametrilor
                    \item Raportarea doar a ajustărilor reușite constituie \textbf{bias de selecție}, nu evidență
                    \item Valorile $p$ trebuie interpretate cu precauție în condițiile \textbf{testării multiple}
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Protocol empiric minim}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Înainte de a revendica un semnal de bulă speculativă}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Ajustarea LPPL pe $\geq 3$ ferestre imbricate
                    \item Verificarea că estimările $t_c$ se grupează
                    \item Verificarea ratei de trecere a filtrelor $> 50\%$
                    \item Calculul IC bootstrap ($N \geq 200$)
                    \item Testarea pe o perioadă cunoscută fără bulă speculativă
                    \item Raportarea \textbf{tuturor} ajustărilor, nu doar a celei mai bune
                    \item Comparația cu modele de referință
                    \item Rularea testului placebo cu randamente amestecate
                    \item Dacă se produc prognoze, compararea erorilor de prognoză folosind Diebold--Mariano
                \end{enumerate}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Capcane frecvente}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Selecția celei mai bune ferestre de ajustare
                    \item Raportarea doar a detecțiilor reușite
                    \item Ignorarea fals pozitivilor și a aproape-reușitelor
                    \item Echivalarea semnalului LPPL cu certitudinea crahului
                    \item Omiterea comparației cu referințe
                    \item Nedeclararea faptului că analiza este \textbf{ex-post}
                    \item Necorectarea pentru testare multiplă
                    \item Raportarea ``LPPL a detectat bula'' fără \textbf{evidență de fals pozitiv}
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

                        \item Creare sintetică: $P^*_i = e^{\hat{P}_i + e^*_i}$
                        \item Re-ajustare LPPL, aplicarea celor 8 filtre
                        \item Stocarea $t_c^{(k)}$ dacă este valid
                    \end{itemize}
                    \item IC: $[t_c^{(2{,}5\%)},\; t_c^{(97{,}5\%)}]$

\begin{frame}{Construirea indicatorului de încredere}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Abordarea multi-scală}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Ajustare LPPL pe \textbf{mai multe ferestre} (variind începutul/sfârșitul)
                    \item Se păstrează doar ajustările care trec \textbf{toate cele 8 filtre}
                    \item Se calculează fracția de ajustări valide $\to$ CI
                \end{enumerate}
            \end{block}
            \vspace{-1mm}
            \begin{exampleblock}{Grila de ferestre}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_1 \in [t - 750, t - 30]$ (date de început)
                    \item $t_2 = t$ (se termină la data curentă)
                    \item Interval fereastră: 30--750 zile
                    \item Grilă: $\sim$100 ferestre pe zi
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{Definiția indicatorului de încredere}
                \vspace{-2mm}
                $$\text{CI}(t) = \frac{\#\{\text{ajustări LPPL valide la }t\}}{\#\{\text{toate ferestrele ajustate la }t\}}$$
                \vspace{-3mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item CI $< 0.3$: Nu se detectează structură LPPL
                    \item CI $\in [0.3, 0.5]$: Ridicat --- monitorizare
                    \item CI $\in [0.5, 0.7]$: Structură LPPL puternică
                    \item CI $> 0.7$: Semnal robust de regim de bulă speculativă
                \end{itemize}
            \end{block}
            \vspace{-1mm}
            \begin{block}{Interpretare}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item CI \textbf{nu} este o probabilitate calibrată de crah --- este un scor de consens al modelului
                    \item Un CI \textbf{în creștere} este mai informativ decât o singură valoare ridicată
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{LPPLS multi-scală (MS-LPPLS)}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Idee cheie {\scriptsize[\href{https://doi.org/10.1080/14697688.2016.1231417}{\textcolor{white}{Demos \& Sornette, 2017}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Ajustare LPPL pe \textbf{ferestre multiple} la diferite scări:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item \textbf{Scurtă} ($<90$d): exuberanță locală
                        \item \textbf{Medie} ($90$--$365$d): bule speculative în dezvoltare
                        \item \textbf{Lungă} ($>365$d): schimbări macro de regim
                    \end{itemize}
                \end{itemize}
            \end{block}
            \vspace{-2mm}
            \begin{exampleblock}{Avantaje}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Captează bule speculative imbricate (bulă speculativă în interiorul bulei speculative)
                    \item Reduce falsurile pozitive prin acord multi-scală
                    \item Estimare mai robustă a lui $t_c$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Indicatorul de încredere multi-scală}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Agregare pe scări $s \in \{$scurtă, medie, lungă$\}$:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $\text{CI}_{\text{MS}}(t) = \frac{1}{3}\sum_{s}\text{CI}_s(t)$
                        \item Avertizare de crah necesită \textbf{aliniere} pe $\geq 2$ scări
                    \end{itemize}
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Implementare}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Grilă de ferestre: $t_1 \in [t - 750,\, t - 30]$
                    \item Ajustare LPPL pe fiecare pereche $(t_1, t_2)$
                    \item Aplicare condiții de filtrare per fereastră
                    \item Grupare ajustări calificate pe scară
                    \item Calculare CI specific scării
                \end{enumerate}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Anti-bule speculative și bule speculative negative}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Definiție {\scriptsize[\href{https://doi.org/10.1080/14697688.2002.0000014}{\textcolor{white}{Sornette \& Zhou, 2002}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Anti-bulă speculativă}: declin super-exponențial cu oscilații log-periodice, terminându-se într-un \textbf{raliu}
                    \item $\ln P(t) = A + B'(t - t_c')^m[1 + C'\cos(\omega\ln(t - t_c') + \phi')]$
                    \item $t > t_c'$ (post-critic), $B' > 0$ (declin spre $A$)
                \end{itemize}
            \end{block}
            \vspace{-1mm}
            \begin{exampleblock}{Diferența cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Bulă speculativă: $t < t_c$, prețul accelerează \textbf{în sus}
                    \item Anti-bulă speculativă: $t > t_c'$, prețul decelerează \textbf{în jos}
                    \item Ambele prezintă oscilații log-periodice
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{Exemple istorice}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item \textbf{Japonia 1990--2003:} anti-bulă Nikkei după vârful din 1989, durând $>$13 ani
                    \item \textbf{S\&P 500 2000--2003:} declin post-dot-com cu log-periodicitate clară
                    \item \textbf{Petrol 2008--2009:} \$147 $\to$ \$32, tipar anti-bulă
                \end{itemize}
            \end{block}

            \begin{block}{Implicații practice}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Detectarea \textbf{punctelor de capitulare} (sfârșitul piețelor bear)
                    \item Semnalarea tranzițiilor de regim spre recuperare
                    \item Cadru simetric: același cod, semn inversat
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Estimarea bayesiană LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Motivație}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL clasic folosește estimări punctuale; \textbf{LPPL bayesian} [Brée \& Joseph, 2013; Geraskin \& Fantazzini, 2013] oferă:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Posterioara completă $p(t_c, m, \omega \mid \text{date})$
                        \item Intervale de credibilitate (nu bootstrap)
                        \item Compararea modelelor prin factori Bayes
                    \end{itemize}
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Specificarea priorilor}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Priori slab informative:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $m \sim \text{Uniform}(0.1, 0.9)$
                        \item $\omega \sim \text{Uniform}(4, 25)$
                        \item $t_c \sim \text{Uniform}(t_N, t_N + 0.5T)$
                        \item $\sigma \sim \text{Half-Cauchy}(0, 1)$
                    \end{itemize}
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Implementare MCMC (HMC)}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Definire verosimilitate: $\varepsilon_t \sim \mathcal{N}(0, \sigma^2)$
                    \item Stabilire priori pe $(t_c, m, \omega, A, B, C_1, C_2)$
                    \item Rulare HMC cu 4 lanțuri, 2000 încălzire
                    \item Verificare convergență ($\hat{R} < 1.01$)
                \end{enumerate}
            \end{alertblock}

            \begin{block}{Avantaje față de bootstrap}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Încorporează cunoștințe anterioare
                    \item Gestionează multimodalitatea natural
                    \item Oferă evidență de model pentru comparare
                    \item Instrumente: \texttt{PyMC}, \texttt{Stan}, \texttt{NumPyro}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }

\begin{frame}{LPPL cu schimbare de regim}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Cadru {\scriptsize[\href{https://doi.org/10.1016/j.physa.2020.125367}{\textcolor{white}{Ardila-Alvarez et al., 2021}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL + \textbf{Markov-Switching}; $S_t \in \{$Normal, Bulă speculativă, Post-crah$\}$
                \end{itemize}
                \vspace{-2mm}
                $$\ln P_t = \begin{cases}
                    \mu + \sigma\varepsilon_t & S_t = \text{Normal} \\
                    \text{LPPL}(t) + \sigma_b\varepsilon_t & S_t = \text{Bulă speculativă} \\
                    \alpha + \beta t + \sigma_c\varepsilon_t & S_t = \text{Post}
                \end{cases}$$
            \end{block}
            \vspace{-2mm}
            \begin{exampleblock}{Matricea de tranziție}
                \vspace{-2mm}
                $$P = \begin{pmatrix}
                    p_{NN} & p_{NB} & 0 \\
                    0 & p_{BB} & p_{BP} \\
                    p_{PN} & 0 & p_{PP}
                \end{pmatrix}$$
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Fără tranziție directă Normal $\to$ Post-crah
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Inovația cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Detectare endogenă a regimului:} Nu este necesar să se pre-specifice ferestrele de bulă speculativă
                    \item \textbf{Probabilități filtrate:} $P(S_t = \text{Bulă speculativă} \mid \mathcal{F}_t)$ în timp real
                    \item \textbf{Modelarea duratei:} Timpul așteptat în starea de bulă speculativă
                \end{itemize}
            \end{exampleblock}
            \vspace{-2mm}
            \begin{block}{Estimare}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Algoritmul EM pentru parametrii de regim
                    \item Optimizare LPPL imbricată în starea de bulă speculativă
                    \item Filtrul Hamilton pentru probabilitățile stărilor
                    \item AIC/BIC pentru numărul de regimuri
                \end{enumerate}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Critici academice}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Preocupări statistice {\scriptsize[\href{https://doi.org/10.1080/14697688.2011.607467}{\textcolor{white}{Bree et al., 2013}}]}}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Supra-ajustare:} 7 parametri pentru $\sim$100 puncte
                    \item \textbf{Bias de anticipare:} ușor de ajustat ex-post
                    \item \textbf{Bias de selecție:} doar succesele sunt raportate
                    \item \textbf{Testare multiplă:} multe ferestre, unele se vor ajusta
                    \item \textbf{Data snooping:} parametri calibrați pe crahuri cunoscute
                \end{enumerate}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Răspuns la critici}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Filtre stricte (toate 8) + evaluare out-of-sample
                    \item Raportare CI, rate de fals pozitiv, teste placebo
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Preocupări empirice}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Falsurile pozitive apar
                    \item Incertitudinea de sincronizare este mare
                    \item Nu toate bulele speculative prezintă tipare LPPL
                    \item Nu toate ajustările LPPL conduc la crahuri
                    \item Parametrii variază între piețe
                \end{enumerate}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Evaluarea onestă}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL este un \textbf{termometru de febră}: semnal ridicat $\Rightarrow$ risc ridicat
                    \item Nu garantează crah; se folosește ca un input printre multe altele
                    \item Umilința este esențială
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Când semnalele LPPL eșuează}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Moduri de eșec}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Piața intră într-o \textbf{corecție laterală} în loc să se prăbușească
                    \item $t_c$ estimat \textbf{se deplasează înainte} pe măsură ce sosesc date noi
                    \item Bula \textbf{se dezumflă treptat} (fără crah brusc)
                    \item Semnalul apare doar pentru \textbf{ferestre înguste}
                    \item Log-periodicitatea poate fi un \textbf{artefact de ajustare}
                    \item $R^2$ ridicat nu dovedește \textbf{validitate predictivă}
                \end{enumerate}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Falsuri pozitive istorice}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Acțiuni americane 2014--2015: semnale LPPL, fără crah
                    \item Bitcoin 2019: semnal LPPL scurt, declin treptat
                    \item Datoria europeană 2011: politică exogenă, nu bulă speculativă LPPL
                \end{itemize}
            \end{alertblock}

            \begin{block}{Interpretare onestă}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Un fals pozitiv poate indica totuși supraîncălzire sau fragilitate
                    \item Dar \textbf{trebuie raportat onest}
                    \item Nu ascundeți niciodată eșecurile în spatele studiilor de caz de succes
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Semnale ambigue: Când avertismentele nu devin crahuri}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Cum pot fi avertismentele LPPL ``corecte'' dar nu utile}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Piața intră într-o \textbf{corecție laterală} în loc să se prăbușească
                    \item $t_c$ estimat \textbf{se deplasează înainte} pe măsură ce sosesc date noi (``$t_c$ rulant'')
                    \item Bula \textbf{se dezumflă treptat} în loc să explodeze (scurgere lentă de aer)
                    \item Doar un \textbf{subset îngust} de ferestre trece filtrele
                    \item Oscilațiile aparente \textbf{dispar} la testarea placebo
                    \item Semnalul este valid ca \textbf{evidență de fragilitate} dar invalid ca semnal de sincronizare a crahului
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Exemplu: S\&P 500, 2014--2015}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPLS CI a crescut peste 0.5 la sfârșitul lui 2014
                    \item Piața a continuat să crească 6 luni
                    \item August 2015: corecție de $-12\%$ (nu un crah)
                    \item Semnalul LPPL a fost consistent cu supraîncălzirea dar sincronizarea a fost decalată cu luni
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{exampleblock}{Mesaj cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Un fals pozitiv nu este neapărat inutil: poate indica supraîncălzire sau fragilitate
                    \item Dar trebuie \textbf{raportat onest} și separat de avertismentele reușite
                    \item Întrebarea științifică nu este ``A existat un crah?'' ci ``A fost regimul consistent cu instabilitatea endogenă?''
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Ce poate și ce nu poate face LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{LPPL \textbf{poate}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Detecta accelerarea super-exponențială a prețului
                    \item Estima cel mai probabil sfârșit al regimului de bulă speculativă
                    \item Oferi ferestre probabilistice de risc
                    \item Cuantifica oscilațiile log-periodice
                    \item Servi ca un input într-un cadru multi-indicator
                    \item Identifica tipare non-bulă speculativă (control negativ)
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{LPPL \textbf{nu poate}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Determina data exactă sau magnitudinea unui crah
                    \item Garanta că o ruptură de regim va avea loc
                    \item Detecta șocuri exogene (pandemii, războaie)
                    \item Funcționa fiabil pe ferestre scurte ($< 100$ zile)
                    \item Înlocui analiza fundamentală sau due diligence
                    \item Distinge bulele speculative de schimbările autentice de regim
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-1mm}
    \begin{block}{Concluzia finală}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item LPPL este un \textbf{instrument de diagnostic}, nu un glob de cristal
            \item Identifică un tipar statistic consistent cu dinamica endogenă a bulelor speculative
            \item Un semnal pozitiv indică instabilitate ridicată; nu face un crah cert
        \end{itemize}
    \end{block}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Temă de master: Diagnosticul LPPL de bulă speculativă}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Sarcină}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Alegeți un activ și o perioadă istorică (bulă speculativă sau non-bulă speculativă)
                    \item Realizați un diagnostic LPPL complet
                \end{itemize}
            \end{block}
            \vspace{-2mm}
            \begin{exampleblock}{Pași obligatorii}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Descărcați prețuri zilnice; transformați în log-prețuri
                    \item Ajustați modele exponențial, cuadratic și LPPL
                    \item Estimați LPPL prin linearizare parțială + DE
                    \item Aplicați cele 8 condiții de filtrare
                    \item Bootstrap CI pentru $t_c$ ($N \geq 200$)
                    \item LPPLS CI pe $\geq 3$ ferestre imbricate
                    \item Comparați cu modelele de referință
                \end{enumerate}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Livrabil}
                \begin{itemize}\setlength{\itemsep}{0pt}

                        \item Incertitudine (CI, sensibilitate la fereastră)
                        \item Rezultate comparație cu referință
                        \item Evaluarea riscului de fals pozitiv
                        \item Implicații pentru managementul riscului
                    \end{itemize}
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Criterii de notare}

                    \item Robustețe și comparație cu referință: 25\%

\section{Concluzii}

\begin{frame}{Rezumat: Concluzii cheie}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Fundamentare teoretică}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Bulele speculative prezintă creștere \textbf{super-exponențială}
                    \item LPPL interpretează bulele speculative ca \textbf{comportament de turmă} + feedback pozitiv
                    \item Prezintă oscilații \textbf{log-periodice}
                    \item Aceeași fizică ca \textbf{tranzițiile de fază}
                \end{enumerate}
            \end{block}
            \vspace{-3mm}
            \begin{exampleblock}{Modelul LPPL}
                $$\ln P = A + B(t_c-t)^m + C(t_c-t)^m\cos(\omega\ln(t_c-t)-\phi)$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item 7 parametri (3 liniari, 4 neliniari)
                    \item Linearizare parțială + estimare DE
                    \item 8 condiții de filtrare pentru validitate
                    \item \textbf{Bootstrap CI} pentru incertitudine
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Rezultate studii de caz (Eroare medie: 10.2z)}
                {\scriptsize
                \vspace{-1mm}
                \begin{center}
                    \begin{tabular}{lcc}
                        \toprule
                        \textbf{Caz} & \textbf{Eroare} & \textbf{95\% CI} \\
                        \midrule
                        Bitcoin 2021 & 2z & [478, 485] \\
                        Bitcoin 2017 & 1z & [167, 169] \\
                        NASDAQ 2000 & 9z & [369, 383] \\
                        Shanghai 2015 & 27z & [169, 187] \\
                        Petrol 2008 & 12z & [382, 423] \\
                        COVID 2020 & N/A & Fără bulă speculativă \\
                        \bottomrule
                    \end{tabular}
                \end{center}
                }
            \end{exampleblock}
            \vspace{-2mm}
            \begin{block}{Mesaj cheie}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL + Bootstrap CI = Diagnostic structurat al regimului de bulă speculativă cu margini de incertitudine
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Concluzii curs}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{block}{Mesajul central}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item LPPL este un \textbf{cadru de diagnostic structurat} pentru identificarea regimurilor endogene de bulă speculativă
            \item Folosit responsabil $\Rightarrow$ sisteme de avertizare timpurie, stress testing, revizuire de risc
            \item Folosit mecanic $\Rightarrow$ supra-ajustare, bias de selecție, încredere falsă
        \end{itemize}
    \end{block}
    \vspace{-3mm}
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Întrebarea științifică}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Nu}: ``A prezis LPPL crahul?''
                    \item \textbf{Ci}: ``A oferit LPPL evidență robustă, ajustată la referință, a unui regim instabil de bulă speculativă?''
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Ce ar trebui să poată face studenții}

                    \item \textbf{Valida} cu 8 filtre + diagnostice reziduale
                    \item \textbf{Compara} cu modele mai simple
                    \item \textbf{Testa} cu bootstrap, placebo, controale
                    \item \textbf{Interpreta} critic --- evidență vs.\ supra-ajustare
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
