\begin{frame}{Răspuns Întrebarea 3}
    \begin{cminipage}{0.95\textwidth}
    \begin{exampleblock}{Răspuns Corect: (B)}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{Pozitiv-definirea matricei $\mathbf{H}_t$}
        \end{itemize}
    \end{exampleblock}

    \pause
    \begin{block}{Explicație}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item BEKK: $\mathbf{H}_t = \mathbf{C}\mathbf{C}' + \mathbf{A}'\boldsymbol{\varepsilon}_{t-1}\boldsymbol{\varepsilon}_{t-1}'\mathbf{A} + \mathbf{B}'\mathbf{H}_{t-1}\mathbf{B}$
            \item $\mathbf{C}\mathbf{C}'$ este p.d.\ (dacă $\mathbf{C}$ are rang plin)
            \item $\mathbf{A}'\mathbf{M}\mathbf{A}$ este p.s.d.\ pentru orice $\mathbf{M}$ p.s.d.
            \item Suma de matrice p.d.\ și p.s.d.\ este p.d. $\Rightarrow$ $\mathbf{H}_t$ p.d.\ automat!
            \item \textbf{Atenție}: stationaritatea necesită o condiție separată pe raza spectrală
        \end{itemize}
    \end{block}
    \end{cminipage}

\begin{frame}{Răspuns Întrebarea 4}

\section{Întrebări Adevărat/Fals}

\begin{frame}{Adevărat sau Fals?}
    \begin{cminipage}{0.95\textwidth}
    \begin{enumerate}\setlength{\itemsep}{0pt}
        \item Modelul CCC presupune că corelațiile între active se schimbă în timp.

        \vspace{0.3cm}

        \item Estimarea DCC în două etape este consistentă dar nu pe deplin eficientă.

        \vspace{0.3cm}

        \item EWMA multivariat este echivalent cu BEKK Scalar cu $\omega = 0$ și $\alpha + \beta = 1$.

        \vspace{0.3cm}

        \item Modelul BEKK diagonal captează propagarea volatilității între active.

        \vspace{0.3cm}

        \item aDCC adaugă doar un parametru suplimentar față de DCC.

        \vspace{0.3cm}

        \item VaR-ul de portofoliu bazat pe DCC este mereu mai mare decât cel bazat pe CCC.

        \vspace{0.3cm}

        \item GO-GARCH funcționează prin descompunerea randamentelor în factori necorelați.

        \vspace{0.3cm}

        \item Corelațiile între acțiuni scad de obicei în perioadele de criză financiară.
    \end{enumerate}
    \end{cminipage}
\end{frame}

\begin{frame}{Răspunsuri Adevărat/Fals}
    \begin{cminipage}{0.95\textwidth}
    \begin{enumerate}\setlength{\itemsep}{0pt}
        \item \textbf{FALS} --- CCC = Constant Conditional Correlation. $\mathbf{R}$ este fix.

        \vspace{0.2cm}

        \item \textbf{ADEVĂRAT} --- Consistent prin QML. Erori standard via sandwich estimator.

        \vspace{0.2cm}

        \item \textbf{ADEVĂRAT} --- $\mathbf{H}_t = (1-\lambda)\boldsymbol{\varepsilon}_{t-1}\boldsymbol{\varepsilon}_{t-1}' + \lambda\mathbf{H}_{t-1}$ cu $\lambda = 0.94$.

        \vspace{0.2cm}

        \item \textbf{FALS} --- BEKK diagonal: $\mathbf{A}$, $\mathbf{B}$ diagonale, deci \textbf{fără spillovers}. Doar BEKK complet captează spillovers.

        \vspace{0.2cm}

        \item \textbf{ADEVĂRAT} --- Parametrul $g$ pentru asimetria negativă. Include DCC ($g=0$) ca caz particular.

        \vspace{0.2cm}

        \item \textbf{FALS} --- VaR-ul DCC este mai mare \textbf{în crize} (corelații crescute), dar poate fi mai mic în perioade calme (corelații scăzute). Depinde de regim.

        \vspace{0.2cm}

        \item \textbf{ADEVĂRAT} --- Folosește o matrice de mixare $\mathbf{Z}$ și factori cu GARCH individual.

        \vspace{0.2cm}

        \item \textbf{FALS} --- Corelațiile \textbf{cresc} în crize (contagiune). Diversificarea eșuează.
    \end{enumerate}
    \end{cminipage}
\end{frame}

\section{Probleme Practice}

\begin{frame}{Problema 1: VaR de portofoliu --- CCC vs DCC}
    \begin{cminipage}{0.95\textwidth}
    \begin{block}{Enunț}
        Un portofoliu egal ponderat ($w_1 = w_2 = 0.5$) conține două active cu:
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item Volatilități zilnice: $\sigma_{1,t} = 2\%$, $\sigma_{2,t} = 3\%$
            \item Corelație CCC: $\rho = 0.4$
            \item Corelație DCC (în criză): $\rho_t = 0.8$
        \end{itemize}

        \vspace{0.3cm}
        Calculați:
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{(a)} Volatilitatea portofoliului cu CCC
            \item \textbf{(b)} Volatilitatea portofoliului cu DCC (în criză)
            \item \textbf{(c)} VaR la 99\% (1 zi) pentru 1.000.000 EUR, ambele modele
            \item \textbf{(d)} Cu cât subestimează CCC riscul în criză?
        \end{itemize}
    \end{block}
    \end{cminipage}
\end{frame}

\begin{frame}{Soluție Problema 1}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{exampleblock}{Răspunsuri}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{(a)} $\sigma_p^2 = 0.25 \times 0.04 + 0.25 \times 0.09 + 2 \times 0.25 \times 0.4 \times 0.02 \times 0.03 = 0.0445\%$

            $\sigma_p = \sqrt{0.000445} = 2.11\%$

            \vspace{0.15cm}

            \item \textbf{(b)} Cu $\rho_t = 0.8$: $\sigma_p^2 = 0.0325 + 0.012 = 0.000565$

            $\sigma_p = \sqrt{0.000565} = 2.38\%$

            \vspace{0.15cm}

            \item \textbf{(c)} VaR 99\% ($z_{0.01} = 2.326$):
            \begin{itemize}\setlength{\itemsep}{0pt}
                \item CCC: $2.326 \times 0.0211 \times 1{,}000{,}000 = \textbf{49{,}069}$ EUR
                \item DCC: $2.326 \times 0.0238 \times 1{,}000{,}000 = \textbf{55{,}359}$ EUR
            \end{itemize}

            \vspace{0.15cm}

            \item \textbf{(d)} CCC subestimează VaR cu $(55{,}359 - 49{,}069)/49{,}069 = \textbf{12.8\%}$ în criză!
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
    \quantlet{TSA\_ch5b\_var\_ccc\_dcc}{https://github.com/QuantLet/TSA/tree/main/TSA_ch5b/TSA_ch5b_var_ccc_dcc}
\end{frame}

\begin{frame}{Problema 2: Hedging dinamic cu DCC}
    \begin{cminipage}{0.95\textwidth}
    \begin{block}{Enunț}
        Un fond deține acțiuni și le acoperă cu futures pe indice. La momentul $t$:
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $\sigma_{\text{spot},t} = 1.5\%$, $\sigma_{\text{futures},t} = 1.2\%$
            \item CCC: $\rho = 0.90$
            \item DCC (criză): $\rho_t = 0.97$
        \end{itemize}

        \vspace{0.3cm}
        Calculați:
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{(a)} Raportul de hedging CCC: $h^* = \rho \cdot \sigma_{\text{spot}} / \sigma_{\text{futures}}$
            \item \textbf{(b)} Raportul de hedging DCC (criză)
            \item \textbf{(c)} Eficacitatea hedging-ului ($HE$) pentru fiecare model
        \end{itemize}
    \end{block}
    \end{cminipage}
\end{frame}

\begin{frame}{Soluție Problema 2}
    \begin{cminipage}{0.95\textwidth}
    \begin{exampleblock}{Răspunsuri}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{(a)} CCC: $h^* = 0.90 \times 1.5/1.2 = \textbf{1.125}$

            \vspace{0.2cm}

            \item \textbf{(b)} DCC: $h_t^* = 0.97 \times 1.5/1.2 = \textbf{1.213}$

            \vspace{0.2cm}

            \item \textbf{(c)} Eficacitatea: $HE = \rho^2$
            \begin{itemize}\setlength{\itemsep}{0pt}
                \item CCC: $HE = 0.90^2 = 81.0\%$ din varianță eliminată
                \item DCC: $HE = 0.97^2 = 94.1\%$ din varianță eliminată
            \end{itemize}

            \vspace{0.2cm}

            CCC sub-acoperă cu $1.213 - 1.125 = 0.088$ unități futures. Expunerea reziduală este semnificativă!
        \end{itemize}
    \end{exampleblock}

    \begin{alertblock}{Concluzie}
        DCC ajustează automat raportul de hedging la condițiile de piață. CCC ratează creșterea corelației în criză.
    \end{alertblock}
    \end{cminipage}
    \quantlet{TSA\_ch5b\_hedge\_ratio}{https://github.com/QuantLet/TSA/tree/main/TSA_ch5b/TSA_ch5b_hedge_ratio}
\end{frame}

\begin{frame}{Problema 3: Selecția modelului}
    \begin{cminipage}{0.95\textwidth}
    \begin{block}{Enunț}
        Un cercetător estimează modele pentru un portofoliu de 3 active:

        \vspace{0.3cm}
        \begin{center}
        {\footnotesize
            \begin{tabular}{lcccc}
                \toprule
                \textbf{Model} & \textbf{Parametri} & \textbf{Log-Lik} & \textbf{AIC} & \textbf{$\rho$ dinamic?} \\
                \midrule
                CCC & 12 & $-3{,}210$ & $6{,}444$ & Nu \\
                DCC & 14 & $-3{,}185$ & $6{,}398$ & Da \\
                aDCC & 15 & $-3{,}180$ & $6{,}390$ & Da \\
                BEKK Diagonal & 12 & $-3{,}200$ & $6{,}424$ & Da \\
                BEKK Complet & 24 & $-3{,}170$ & $6{,}388$ & Da \\
                \bottomrule
            \end{tabular}
        }
        \end{center}

        \vspace{0.3cm}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{(a)} Care model are AIC-ul cel mai bun?
            \item \textbf{(b)} Puteți testa CCC vs DCC cu un test LR? Calculați statistica.
            \item \textbf{(c)} Ce model recomandați și de ce?
        \end{itemize}
    \end{block}
    \end{cminipage}
\end{frame}

\begin{frame}{Soluție Problema 3}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{exampleblock}{Răspunsuri}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item \textbf{(a)} BEKK complet are cel mai mic AIC ($6{,}388$), urmat de aDCC ($6{,}390$).

            \vspace{0.15cm}

            \item \textbf{(b)} Da, CCC este imbricat în DCC ($a = b = 0$):

            $LR = 2 \times (-3{,}185 - (-3{,}210)) = 2 \times 25 = 50$

            Sub $H_0$: $LR \sim \chi^2(2)$. Valoare critică la 1\%: 9.21.

            $50 \gg 9.21$ $\Rightarrow$ \textbf{CCC puternic respins}!

            \vspace{0.15cm}

            \item \textbf{(c)} \textbf{aDCC}: AIC aproape de BEKK complet dar cu 9 parametri mai puțini. Pentru $N = 3$, BEKK complet este fezabil, dar aDCC oferă un compromis mai bun între flexibilitate și parsimonie. Dacă spillovers sunt importante $\Rightarrow$ BEKK complet.
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}

from scipy.optimize import minimize
def dcc_loglik(params, z, Qbar):
    a, b = params
    T, N = z.shape
    Qt, ll, rho_t = Qbar.copy(), 0, np.zeros(T)
    for t in range(1, T):
        Qt = (1-a-b)*Qbar + a*np.outer(z[t-1], z[t-1]) + b*Qt
        Rt = Qt / np.sqrt(np.outer(np.diag(Qt), np.diag(Qt)))
        rho_t[t] = Rt[0, 1]
        ll += np.log(np.linalg.det(Rt)) + z[t]@np.linalg.inv(Rt)@z[t] - z[t]@z[t]
    return 0.5 * ll, rho_t
res = minimize(lambda p: dcc_loglik(p, z, Qbar)[0],
    x0=[0.02, 0.95], bounds=[(0,0.3),(0,0.999)],
    constraints={'type':'ineq','fun': lambda p: 0.999-p[0]-p[1]})
a_hat, b_hat = res.x
_, rho_dcc = dcc_loglik([a_hat, b_hat], z, Qbar)
\end{lstlisting}
