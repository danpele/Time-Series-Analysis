            \item The Efficient Market Hypothesis says crashes are \textbf{unpredictable} random shocks

\begin{frame}{Understanding Super-Exponential Growth}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Normal Exponential Growth}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Standard asset pricing: $P(t) = P_0 e^{rt}$
                    \item Growth rate: $\frac{d\ln P}{dt} = r$ (\textbf{constant})
                    \item Price doubles every $T_2 = \ln 2 / r$ periods
                    \item \textbf{Sustainable indefinitely}
                    \item Describes long-term market returns
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Example: S\&P 500}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Historical return $\approx$ 10\%/year
                    \item Doubling time $\approx$ 7 years
                    \item Consistent growth for 100+ years
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Super-Exponential (Bubble) Growth}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item During bubbles: $P(t) \sim (t_c - t)^{-\alpha}$, $\alpha > 0$
                    \item Growth rate: $\frac{d\ln P}{dt} = \frac{\alpha}{t_c - t}$ (\textbf{accelerating!})
                    \item Rate $\to \infty$ as $t \to t_c$
                    \item \textbf{Must end at finite time $t_c$}
                    \item Describes bubble phase
                \end{itemize}
            \end{alertblock}

            \begin{block}{Key Intuition}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item In a bubble, the \textit{rate of rise} keeps increasing
                    \item This acceleration is \textbf{unsustainable}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Why Does Super-Exponential Growth Occur?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Positive Feedback Loops}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Price rises attract \textbf{new buyers}
                    \item New buyers push price \textbf{higher}
                    \item Higher prices attract \textbf{more buyers}
                    \item Cycle \textbf{accelerates}
                    \item Opposite of negative feedback (mean reversion) that EMH assumes
                \end{enumerate}
            \end{block}

            \begin{alertblock}{Mechanisms}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Herding:} Following the crowd
                    \item \textbf{FOMO:} Fear of missing out
                    \item \textbf{Leverage:} Rising prices $\to$ more collateral $\to$ more buying
                    \item \textbf{Media:} Success stories attract attention
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Mathematical Formulation}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Demand depends on past returns: $D(t) = D_0 + \gamma \cdot \text{Return}_{t-1}$
                    \item If $\gamma > 0$ (trend following): $\frac{dP}{dt} \propto D(t) \propto \frac{dP}{dt}\big|_{t-1}$
                    \item Result: $\frac{d^2 \ln P}{dt^2} > 0$ (Super-exponential!)
                \end{itemize}
            \end{exampleblock}

            \begin{block}{The Endgame}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Super-exponential growth \textbf{cannot continue forever}
                    \item At $t_c$: run out of new buyers
                    \item Hit a constraint (credit, cash)
                    \item Experience a trigger event
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{The Ising Model: Simplest Collective Behavior}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Physical Setup {\scriptsize[\href{https://doi.org/10.1007/BF02980577}{\textcolor{white}{Ising, 1925}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Each site $i$ has spin $s_i \in \{-1, +1\}$
                    \item Neighboring spins interact with strength $J$
                    \item External magnetic field $h$ can bias spins
                    \item System in thermal equilibrium at temperature $T$
                \end{itemize}
            \end{block}

            \begin{alertblock}{The Hamiltonian}
                $$H = -J \sum_{\langle i,j \rangle} s_i s_j - h \sum_i s_i$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item First sum over neighbor pairs
                    \item $J > 0$: aligned spins $=$ low energy
                    \item $h > 0$: up spins $=$ low energy
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Financial Interpretation {\scriptsize[\href{https://doi.org/10.1017/S1365100500015029}{\textcolor{white}{Cont \& Bouchaud, 2000}}]}}
                \begin{center}
                    \begin{tabular}{ll}
                        \toprule
                        \textbf{Physics} & \textbf{Finance} \\
                        \midrule
                        Spin up $+1$ & Buy order \\
                        Spin down $-1$ & Sell order \\
                        Coupling $J$ & Herding strength \\
                        Field $h$ & Fundamental signal \\
                        Temperature $T$ & Noise / uncertainty \\
                        Magnetization $m$ & Market sentiment \\
                        \bottomrule
                    \end{tabular}
                \end{center}
            \end{exampleblock}

            \begin{block}{}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item High $J$ (strong herding) + low $T$ (low noise)
                    \item $\Rightarrow$ Traders align $\Rightarrow$ \textbf{Bubble forms!}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{What Are Log-Periodic Oscillations?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{The Pattern}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Compress} in real time (faster near $t_c$)
                    \item Are \textbf{evenly spaced} in $\ln(t_c - t)$
                    \item Modulate the power law trend
                    \item Reflect hierarchical market structure
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Mathematical Form}
                $$\ln P(t) = A + B(t_c-t)^m \cdot [1 + C\cos(\omega\ln(t_c-t))]$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Cosine creates oscillations in $\ln(t_c - t)$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Why ``Log-Periodic''?}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Plot against $\tau = \ln(t_c - t)$: $\ln P = A + Be^{m\tau}[1 + C\cos(\omega\tau)]$
                    \item Oscillations have \textbf{constant period} in log-time
                    \item Period $= 2\pi/\omega$, scaling ratio $\lambda = e^{2\pi/\omega}$
                \end{itemize}
            \end{alertblock}

            \begin{block}{Physical Meaning}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Each oscillation = a ``round'' of the bubble
                    \item Prices accelerate, briefly correct, then accelerate faster
                    \item Each round is $\lambda$ times shorter than previous
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Continuous Scale Invariance}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Definition}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $f(x)$ is scale invariant if: $f(\lambda x) = \lambda^{-\alpha} f(x) \;\forall \lambda > 0$
                    \item Scaling by \textbf{any} factor $\lambda$ just rescales the function
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Unique Solution}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item The \textbf{only} solutions are: $f(x) = C \cdot x^{-\alpha}$
                    \item Pure power laws!
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Examples in Finance}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Return tails:} $P(|r| > x) \sim x^{-\alpha}$
                    \item \textbf{Firm sizes:} Zipf's law
                    \item \textbf{Trading volumes:} Power law distribution
                    \item \textbf{Price impact:} $\Delta P \sim V^{1/2}$
                    \item These arise when there's no characteristic scale
                \end{itemize}
            \end{exampleblock}

            \begin{block}{Limitation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Pure CSI gives only power laws
                    \item Where do the oscillations come from?
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Discrete Scale Invariance}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Definition {\scriptsize[\href{https://doi.org/10.1016/S0370-1573(97)00076-8}{\textcolor{white}{Sornette, 1998}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Invariance only for \textbf{specific} $\lambda_0$: $f(\lambda_0 x) = \lambda_0^{-\alpha} f(x)$
                    \item The system has a \textbf{preferred} scale ratio
                \end{itemize}
            \end{block}

            \begin{block}{General Solution}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item $f(x) = x^{-\alpha} \cdot G(\ln x)$ where $G$ is periodic with period $\ln\lambda_0$
                    \item Fourier: $f(x) = x^{-\alpha}\sum_n c_n e^{in\omega\ln x}$, $\omega = 2\pi/\ln\lambda_0$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Leading Order Approximation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Keep only $n = 0, \pm 1$: $f(x) \approx x^{-\alpha}[A' + B'\cos(\omega\ln x + \phi)]$
                    \item \textbf{Power law + log-periodic oscillations!}
                \end{itemize}
            \end{exampleblock}

            \begin{block}{Physical Interpretation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item DSI arises from \textbf{hierarchical structure}
                    \item Fractal geometry, hierarchical networks
                    \item Multi-scale organization
                    \item Each level related by factor $\lambda_0$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

    \begin{columns}[T]
        \begin{column}{0.47\textwidth}

                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_c$: Critical time (crash window)
                    \item $m$: Power law exponent ($0 < m < 1$)
                    \item $\omega$: Log-periodic frequency

\begin{frame}{Two Equivalent LPPL Parameterizations}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Phase Form (7 parameters)}
                \begin{equation*}
                    \ln P(t) = A + B\tau^m + C\tau^m \cos(\omega \ln\tau - \phi)
                \end{equation*}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $\tau = t_c - t$; parameters: $A, B, C, t_c, m, \omega, \phi$
                    \item Physically interpretable: $C$ = oscillation amplitude, $\phi$ = phase
                    \item Used in theoretical derivations
                    \item 4 nonlinear parameters: $t_c, m, \omega, \phi$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Linearized Form (7 parameters)}
                \begin{equation*}
                    \ln P(t) = A + Bf + C_1 g + C_2 h
                \end{equation*}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $f = \tau^m$, $g = \tau^m\cos(\omega\ln\tau)$, $h = \tau^m\sin(\omega\ln\tau)$
                    \item Parameters: $A, B, C_1, C_2, t_c, m, \omega$
                    \item Only 3 nonlinear: $t_c, m, \omega$
                    \item $A, B, C_1, C_2$ solved by OLS
                    \item Recover: $C = \sqrt{C_1^2 + C_2^2}$, $\phi = \operatorname{atan2}(C_2, C_1)$
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}

    \vspace{-1mm}
    \begin{exampleblock}{Why Two Forms?}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item Linearized form: 4 $\to$ 3 nonlinear parameters, much better convergence
            \item Both forms are mathematically identical --- they fit the same curve
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 1: Rational Bubble Theory}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{No-Arbitrage Condition {\scriptsize[\href{https://doi.org/10.3386/w0945}{\textcolor{white}{Blanchard \& Watson, 1982}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Rational bubble: $\E_t[dP] = P r\, dt + P h(t) \kappa\, dt$
                    \item $r$: Risk-free rate
                    \item $h(t)$: Crash hazard rate (probability/time)
                    \item $\kappa$: Expected crash size (e.g., 30\%)
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Interpretation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Higher crash risk $\Rightarrow$ higher expected return required
                    \item Price must rise faster to compensate
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Key Insight}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item If $h(t)$ \textbf{increases}, expected returns \textbf{accelerate}
                    \item $\frac{d\ln P}{dt} = r + h(t)\kappa$; as $h(t) \uparrow$, growth $\uparrow$
                    \item This creates super-exponential growth!
                \end{itemize}
            \end{alertblock}

            \begin{block}{The Question}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item What functional form should $h(t)$ take?
                    \item Physics provides the answer through critical phenomena!
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 2: The Crash-Compensated Pricing Equation}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Setup}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Price $P(t)$ can crash at random time $t^*$
                    \item At crash: $P \to P(1 - \kappa)$, with $\kappa \in (0,1)$
                    \item Hazard rate: $h(t) = \Pr[\text{crash in } (t, t+dt)] / dt$
                    \item Before the crash, price follows:
                \end{itemize}
                \vspace{1mm}
                $$\frac{dP}{P} = \mu(t)\,dt + \sigma\,dW - \kappa\,dj$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $dj$: jump process with intensity $h(t)$
                    \item $\E[dj] = h(t)\,dt$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{No-Arbitrage Condition}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Expected return must equal $r$ (risk-free rate):
                \end{itemize}
                $$\E\!\left[\frac{dP}{P}\right] = \mu(t)\,dt - \kappa\,h(t)\,dt = r\,dt$$
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Solving for $\mu(t)$:
                \end{itemize}
                $$\boxed{\mu(t) = r + \kappa\, h(t)}$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Higher crash risk $\Rightarrow$ price must rise \textbf{faster}
                    \item Investors demand compensation for bearing crash risk
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 3: From Drift to Log-Price}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{block}{Integrating the Pricing Equation (before the crash)}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item Take expectations and ignore noise ($\sigma\,dW$) for the conditional mean path:
            \vspace{-2mm}
                $$\frac{d\,\E[\ln P]}{dt} = \mu(t) = r + \kappa\, h(t)$$
            \vspace{-4mm}
            \item Integrate from $0$ to $t$:
            \vspace{-2mm}
                $$\E[\ln P(t)] = \ln P(0) + r\,t + \kappa \int_0^t h(s)\,ds$$
            \vspace{-4mm}
            \item The shape of $\ln P(t)$ depends on the \textbf{cumulative hazard} $\int_0^t h(s)\,ds$
            \item If $h(t) = $ constant $\Rightarrow$ exponential growth (EMH)
            \item If $h(t)$ \textbf{increases} $\Rightarrow$ \textbf{super-exponential growth} (bubble!)
        \end{itemize}
    \end{block}

    \vspace{-2mm}

    \begin{alertblock}{The Key Question}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item What functional form should $h(t)$ take during a bubble?
            \item \textbf{Critical phenomena} (Part II--III): near a critical point, quantities follow \textbf{power laws} with \textbf{log-periodic corrections}
        \end{itemize}
    \end{alertblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 4: Why Power-Law Hazard?}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{From Critical Phenomena to Finance}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Near a phase transition at $T_c$, physical quantities diverge as power laws
                    \item Susceptibility: $\chi \sim |T - T_c|^{-\gamma}$
                    \item Correlation length: $\xi \sim |T - T_c|^{-\nu}$
                    \item \textbf{Market analogy:} replace $T \to t$, $T_c \to t_c$
                    \item Crash ``susceptibility'' follows the same critical scaling
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{The Power-Law Hazard Form}
                \vspace{-2mm}
                $$\boxed{h(t) = \alpha(t_c - t)^{m-1}, \quad 0 < m < 1}$$
                \vspace{-3mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Simplest function consistent with critical scaling
                    \item $\alpha > 0$: amplitude of the singularity
                    \item $m$ is the \textbf{critical exponent} of the bubble
                    \item Analogous to universality classes in physics
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-2mm}
    \begin{exampleblock}{Why $0 < m < 1$? Two essential constraints}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $m < 1 \;\Rightarrow\; m{-}1 < 0 \;\Rightarrow\; h(t) \to \infty$ as $t \to t_c$: crash becomes increasingly likely
            \item $m > 0 \;\Rightarrow\; \int_0^{t_c}\! h(s)\,ds < \infty$: cumulative hazard stays finite, so $\ln P$ remains bounded
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 5: Integration of the Power-Law Hazard}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Setting Up the Integral}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item From Step 3: $\ln P(t) = \ln P(0) + rt + \kappa\!\int_0^t\! h(s)\,ds$
                    \item Substitute $h(s) = \alpha(t_c - s)^{m-1}$:
                \end{itemize}
                \vspace{-2mm}
                $$\kappa\!\int_0^t\! h(s)\,ds = \kappa\alpha\!\int_0^t\! (t_c - s)^{m-1}\,ds$$
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Substitution: $u = t_c - s$, $du = -ds$
                    \item Limits: $s{=}0 \!\to\! u{=}t_c$;\; $s{=}t \!\to\! u{=}t_c{-}t$
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Computing the Integral}
                \vspace{-2mm}
                \begin{align*}
                    \kappa\alpha\!\int_0^t\! (t_c{-}s)^{m-1}ds &= \kappa\alpha\!\int_{t_c-t}^{t_c}\! u^{m-1}\,du \\[-2pt]
                    &= \frac{\kappa\alpha}{m}\Bigl[t_c^m - (t_c{-}t)^m\Bigr]
                \end{align*}
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Converges because $m > 0$
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-2mm}
    \begin{exampleblock}{Collecting Terms}
        \vspace{-2mm}
        $$\ln P(t) = \underbrace{\ln P(0) + rt + \tfrac{\kappa\alpha}{m}t_c^m}_{\displaystyle A} + \underbrace{\bigl(-\tfrac{\kappa\alpha}{m}\bigr)}_{\displaystyle B}(t_c - t)^m \qquad\Longrightarrow\qquad \boxed{\ln P(t) = A + B(t_c - t)^m}$$
        \vspace{-3mm}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $B = -\kappa\alpha/m < 0$ because $\kappa, \alpha, m > 0$ --- ensures \textbf{super-exponential} growth toward $t_c$
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 6: The Log-Periodic Oscillatory Integral}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{DSI Modifies the Hazard Rate}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item From DSI (Part III), the hazard acquires oscillations:
                \end{itemize}
                \vspace{-2mm}
                $$h(t) = \alpha(t_c{-}t)^{m-1}\bigl[1 + c\cos(\omega\ln(t_c{-}t) {-} \phi')\bigr]$$
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Step 5 handled the ``$1$'' term $\to$ $B(t_c{-}t)^m$
                    \item Now evaluate the oscillatory correction:
                \end{itemize}
                \vspace{-2mm}
                $$I = \int u^{m-1}\cos(\omega\ln u)\,du$$
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Complex Exponential Trick}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Write $\cos(\omega\ln u) = \operatorname{Re}[u^{i\omega}]$
                    \item So $I = \operatorname{Re}\!\left[\int u^{m-1+i\omega}\,du\right] = \operatorname{Re}\!\left[\frac{u^{m+i\omega}}{m+i\omega}\right]$
                    \item Polar form: $m + i\omega = \sqrt{m^2+\omega^2}\;e^{i\theta}$
                    \item where $\theta = \arctan(\omega/m)$
                \end{itemize}
                \vspace{-2mm}
                $$\boxed{I = \frac{u^m}{\sqrt{m^2+\omega^2}}\cos(\omega\ln u - \theta)}$$
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-1mm}
    \begin{exampleblock}{Combining Power-Law + Oscillatory Terms $\longrightarrow$ The LPPL Equation}
        \vspace{-1mm}
        $$\ln P(t) = A + B(t_c{-}t)^m + C(t_c{-}t)^m\cos\bigl(\omega\ln(t_c{-}t) - \phi\bigr)$$
        \vspace{-3mm}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $C = -\kappa\alpha c\,/\,\sqrt{m^2 + \omega^2}$; \quad $\phi = \phi' + \arctan(\omega/m)$
            \item The phase shift $\phi$ arises naturally from the integration --- it is \textbf{not} a free parameter
        \end{itemize}
    \end{exampleblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Derivation Step 7: The Complete Chain}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{block}{From Rational Expectations to LPPL {\scriptsize[\href{https://doi.org/10.48550/arXiv.cond-mat/9903321}{\textcolor{white}{Johansen et al., 2000}}]}}
        \begin{enumerate}\setlength{\itemsep}{0pt}
            \item \textbf{No-arbitrage} $\Rightarrow$ $\mu(t) = r + \kappa\,h(t)$
            \item \textbf{Integrate} $\Rightarrow$ $\ln P(t) = \ln P(0) + rt + \kappa\!\int_0^t h(s)\,ds$
            \item \textbf{Critical scaling} $\Rightarrow$ $h(t) = \alpha(t_c - t)^{m-1}$, \;$0 < m < 1$
            \item \textbf{Power-law integral} $\Rightarrow$ $\ln P = A + B(t_c-t)^m$, \;$B = -\kappa\alpha/m < 0$
            \item \textbf{DSI oscillations} $\Rightarrow$ $h(t) = \alpha(t_c{-}t)^{m-1}[1 + c\cos(\omega\ln(t_c{-}t) {-} \phi')]$
            \item \textbf{Complex integral} $\Rightarrow$ adds $C(t_c{-}t)^m\cos(\omega\ln(t_c{-}t) - \phi)$
        \end{enumerate}
    \end{block}
    \vspace{-2mm}
    \begin{exampleblock}{}
        \vspace{-1mm}
        \begin{center}
            \large $\boxed{\ln P(t) = A + B(t_c - t)^m + C(t_c - t)^m \cos\bigl(\omega \ln(t_c - t) - \phi\bigr)}$
        \end{center}
        \vspace{-2mm}
    \end{exampleblock}
    \vspace{-2mm}
    \begin{alertblock}{Key constraints from the derivation}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item $B < 0$ (super-exponential growth); \; $0 < m < 1$ (singularity + finite price)
            \item $C = -\kappa\alpha c/\sqrt{m^2{+}\omega^2}$; \; $\phi = \phi' + \arctan(\omega/m)$
        \end{itemize}
    \end{alertblock}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Partial Linearization Method}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Rewrite LPPL (Linearized Form)}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Basis: $f_1 = 1$, $f_2 = (t_c - t)^m$, $f_3 = f_2\cos(\omega\ln(t_c-t))$, $f_4 = f_2\sin(\omega\ln(t_c-t))$
                    \item Then: $\ln P(t) = Af_1 + Bf_2 + C_1 f_3 + C_2 f_4$
                    \item This is \textbf{linear} in $(A, B, C_1, C_2)$!
                \end{itemize}
            \end{block}

            \begin{exampleblock}{OLS Solution}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Given $(t_c, m, \omega)$:
                    \item $(A, B, C_1, C_2)' = (F'F)^{-1}F'\mathbf{y}$ where $\mathbf{y} = \ln P$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Outer Optimization}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Minimize $\text{SSR} = \sum_t [\ln P(t) - \widehat{\ln P}(t)]^2$
                    \item $\widehat{\ln P}$ uses OLS-optimal $(A, B, C_1, C_2)$
                    \item Use \textbf{Differential Evolution}:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Population of candidate solutions
                        \item Mutation and crossover operators
                        \item Converges to global optimum
                    \end{itemize}
                \end{itemize}
            \end{alertblock}

            \begin{block}{Bounds}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_c \in [T, T + 0.5(T - t_1)]$
                    \item $m \in [0.1, 0.9]$, $\omega \in [4, 25]$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

\begin{frame}{Implementation: LPPL Fitting Algorithm}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Algorithm Overview}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Input:} Time index $t$, log-prices $\ln P$
                    \item \textbf{Outer loop:} DE optimizes $(t_c, m, \omega)$
                    \item \textbf{Inner step:} OLS solves $(A, B, C_1, C_2)$ via basis matrix $\mathbf{F}$
                    \item \textbf{Cost:} $\sum_i (\ln P_i - \hat{P}_i)^2$
                    \item \textbf{Output:} 7 optimal parameters
                \end{enumerate}
            \end{block}
            \vspace{-1mm}
            \begin{exampleblock}{Basis Matrix $\mathbf{F}$}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $f_{i1} = 1$ (intercept)
                    \item $f_{i2} = (t_c - t_i)^m$ (power law)
                    \item $f_{i3} = f_{i2}\cos(\omega\ln(t_c - t_i))$
                    \item $f_{i4} = f_{i2}\sin(\omega\ln(t_c - t_i))$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{DE Settings}

\begin{frame}{The Window Selection Problem}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Why Window Matters}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Start date} ($t_1$): When does bubble begin?
                    \item \textbf{End date} ($t_2$): Current observation point
                    \item \textbf{Window length}: $\Delta t = t_2 - t_1$
                \end{itemize}
            \end{block}

            \begin{alertblock}{The Challenge}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Too short: Insufficient data for reliable fit
                    \item Too long: May include pre-bubble regime
                    \item Wrong start: Miss bubble inception or include noise
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Multi-Scale Approach}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item [\href{https://doi.org/10.1142/S2010139215500044}{Sornette et al., 2015}] recommend:
                \end{itemize}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Fit LPPL over \textbf{multiple windows}
                    \item Vary $t_1$ systematically
                    \item Keep $t_2$ fixed (today)
                    \item Window sizes: 60--750 days
                    \item Aggregate results
                \end{enumerate}
            \end{exampleblock}

            \begin{block}{Typical Window Sizes}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Short: 60--120 days
                    \item Medium: 120--250 days
                    \item Long: 250--750 days
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Window Selection: Practical Guidelines}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Minimum Window Requirements}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item At least \textbf{60 observations} for statistical reliability
                    \item Should contain $\geq 2$ log-periodic cycles
                    \item Rule of thumb: $\Delta t > \frac{4\pi}{\omega}$
                    \item For $\omega \approx 6$: minimum $\sim$125 days
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Start Date Selection}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Regime change in volatility
                    \item Beginning of sustained uptrend
                    \item Break from mean-reverting behavior
                    \item News event triggering bubble
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Window Robustness Check}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Pass filters for multiple windows
                    \item Show consistent $t_c$ estimates ($\pm 30$ days)
                    \item Maintain $\omega$ stability ($\pm 20\%$)
                    \item Keep $m$ in valid range across windows
                \end{enumerate}
            \end{alertblock}

            \begin{block}{Red Flags}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_c$ varies wildly with window
                    \item Only passes filters for 1--2 windows
                    \item Parameters at boundary values
                    \item $R^2$ drops sharply for nearby windows
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Preview: From Fit to Validation}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{1. Residual Diagnostics}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Ljung-Box}: $Q = n(n+2)\sum_{k=1}^{h}\frac{\hat{\rho}_k^2}{n-k}$
                    \item \textbf{ARCH-LM test}: No heteroskedasticity
                    \item \textbf{Jarque-Bera}: Normality of residuals
                \end{itemize}
            \end{block}

            \begin{alertblock}{2. Likelihood Ratio Test}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $LR = 2(\ell_{LPPL} - \ell_{null}) \sim \chi^2(df = 4)$
                    \item $H_0$: Exponential model sufficient
                    \item $H_1$: LPPL provides better fit
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{3. Bootstrap Confidence Intervals}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Resample residuals (block bootstrap)
                    \item Re-estimate LPPL 100+ times
                    \item Compute 95\% CI for $t_c$, $m$, $\omega$
                \end{enumerate}
            \end{exampleblock}

            \begin{block}{4. Out-of-Sample Validation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Fit on data up to $t^*$, estimate $t_c$ and price path
                    \item Compare with realized values
                    \item Track estimation error over rolling windows
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Bootstrap Confidence Intervals: Algorithm}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Block Bootstrap Algorithm}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Fit LPPL on original data $\Rightarrow \hat{\theta}$
                    \item Compute residuals: $e_i = \ln P_i - \widehat{\ln P}_i$
                    \item Set block size $b = \lfloor\sqrt{N}\rfloor$
                    \item For $k = 1, \ldots, B$ (e.g., $B = 500$):
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Resample residual blocks with replacement
                        \item Create synthetic prices: $P^*_i = \exp(\widehat{\ln P}_i + e^*_i)$
                        \item Re-fit LPPL $\Rightarrow \hat{\theta}^{(k)}$
                        \item Store $t_c^{(k)}$ if filters pass
                    \end{itemize}
                    \item CI: $[t_c^{(0.025)},\; t_c^{(0.975)}]$
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Why Block Bootstrap?}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item LPPL residuals are \textbf{autocorrelated}
                    \item Simple bootstrap destroys serial dependence
                    \item Block bootstrap preserves local structure
                    \item Block size $b \approx \sqrt{N}$ is standard
                \end{itemize}
            \end{alertblock}

            \begin{exampleblock}{Quality Checks}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Valid bootstrap rate $> 70\%$
                    \item $t_c$ distribution is unimodal
                    \item CI width $< 0.4T$ (not too wide)
                    \item Ex-post diagnostic: actual peak within 95\% CI
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
    }
    \hfill\quantlet{TSA\_ch13\_bootstrap\_ci}{https://github.com/QuantLet/TSA/tree/main/TSA\_ch13/TSA\_ch13\_bootstrap\_ci}

\begin{frame}{Scientific Standard for LPPL Evidence}
    \begin{cminipage}{0.95\textwidth}

    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{What Must Be Reported}

                    \item \textbf{All} failed and successful signals

                    \item LPPLS CI elevated before the peak

                    \item $p$-values must be interpreted cautiously under \textbf{multiple testing}

\begin{frame}{Minimum Empirical Protocol}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Before Claiming a Bubble Signal}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Fit LPPL on $\geq 3$ nested windows
                    \item Check that $t_c$ estimates cluster
                    \item Verify filter pass rate $> 50\%$
                    \item Compute bootstrap CI ($N \geq 200$)
                    \item Test on a known non-bubble period
                    \item Report \textbf{all} fits, not just the best
                    \item Compare against benchmark models
                    \item Run shuffled-return placebo test
                    \item If forecasts are produced, compare forecast errors using Diebold--Mariano
                \end{enumerate}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Common Pitfalls}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item Cherry-picking the best-fitting window
                    \item Reporting only successful detections
                    \item Ignoring false positives and near-misses
                    \item Equating LPPL signal with crash certainty
                    \item Omitting benchmark comparison
                    \item Not disclosing whether analysis is \textbf{ex-post}
                    \item Not correcting for multiple testing
                    \item Reporting ``LPPL detected the bubble'' without \textbf{false-positive evidence}
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    }

\begin{frame}{Case Study: Dot-Com Bubble 2000}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Data}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Asset:} NASDAQ Composite (\texttt{\^{}IXIC})
                    \item \textbf{Window:} Oct 1, 1998 -- Mar 10, 2000
                    \item \textbf{Observations:} 373 trading days
                    \item \textbf{Actual peak:} March 10, 2000
                    \item \textbf{Crash:} $-78\%$ over 30 months
                \end{itemize}
            \end{block}

            \begin{exampleblock}{LPPL Analysis {\scriptsize[\href{https://doi.org/10.48550/arXiv.cond-mat/9903321}{\textcolor{white}{Johansen \& Sornette, 1999}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Published \textbf{January 1999} in \href{https://www.risk.net}{\textit{Risk Magazine}}
                    \item Estimated critical window: Q1 2000
                    \item Ex-ante warning 14 months before peak
                \end{itemize}
            \end{exampleblock}
            \vspace{-2mm}
            {\scriptsize\textcolor{MediumGray}{Mode: \textbf{ex-ante} | Window: pre-specified | Benchmark: no | Bootstrap: yes}}
        \end{column}

\begin{frame}{Step 2: LPPL Fitting via Partial Linearization}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Estimation Procedure}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item \textbf{Fix} nonlinear params $(t_c, m, \omega)$
                    \item \textbf{Compute} basis: $f_i = (t_c - t_i)^m$, $g_i = f_i\cos(\omega\ln(t_c - t_i))$, $h_i = f_i\sin(\omega\ln(t_c - t_i))$
                    \item \textbf{Solve} OLS: $\ln P_i = A + Bf_i + C_1 g_i + C_2 h_i$
                    \item \textbf{Optimize} outer loop via Differential Evolution
                \end{enumerate}
            \end{block}

            \begin{exampleblock}{DE Search Bounds (wider than filters)}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_c \in [t_N + 1,\; t_N + 90]$ days
                    \item $m \in [0.01,\; 0.99]$ (filter: $[0.1, 0.9]$)
                    \item $\omega \in [4.0,\; 25.0]$

\begin{frame}{Building the Confidence Indicator}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{The Multi-Scale Approach}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Fit LPPL on \textbf{many windows} (vary start/end)
                    \item Keep only fits passing \textbf{all 8 filters}
                    \item Count fraction of valid fits $\to$ CI
                \end{enumerate}
            \end{block}
            \vspace{-1mm}
            \begin{exampleblock}{Window Grid}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item $t_1 \in [t - 750, t - 30]$ (start dates)
                    \item $t_2 = t$ (end at current date)
                    \item Window range: 30--750 days
                    \item Grid: $\sim$100 windows per day
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{Confidence Indicator Definition}
                \vspace{-2mm}
                $$\text{CI}(t) = \frac{\#\{\text{valid LPPL fits at }t\}}{\#\{\text{all fitted windows at }t\}}$$
                \vspace{-3mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item CI $< 0.3$: No LPPL structure detected
                    \item CI $\in [0.3, 0.5]$: Elevated --- monitor
                    \item CI $\in [0.5, 0.7]$: Strong LPPL structure
                    \item CI $> 0.7$: Robust bubble-regime signal
                \end{itemize}
            \end{block}
            \vspace{-1mm}
            \begin{block}{Interpretation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item CI is \textbf{not} a calibrated crash probability --- it is a model-consensus score
                    \item A \textbf{rising} CI is more informative than a single high value
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Multi-Scale LPPLS (MS-LPPLS)}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Key Idea {\scriptsize[\href{https://doi.org/10.1080/14697688.2016.1231417}{\textcolor{white}{Demos \& Sornette, 2017}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Fit LPPL across \textbf{multiple windows} at different scales:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item \textbf{Short} ($<90$d): local exuberance
                        \item \textbf{Medium} ($90$--$365$d): developing bubbles
                        \item \textbf{Long} ($>365$d): macro regime shifts
                    \end{itemize}
                \end{itemize}
            \end{block}
            \vspace{-2mm}
            \begin{exampleblock}{Advantages}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Captures nested bubbles (bubble within bubble)
                    \item Reduces false positives via multi-scale agreement
                    \item More robust $t_c$ estimation
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Multi-Scale Confidence Indicator}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Aggregate across scales $s \in \{$short, medium, long$\}$:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $\text{CI}_{\text{MS}}(t) = \frac{1}{3}\sum_{s}\text{CI}_s(t)$
                        \item Crash warning requires \textbf{alignment} across $\geq 2$ scales
                    \end{itemize}
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Implementation}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Window grid: $t_1 \in [t - 750,\, t - 30]$
                    \item Fit LPPL on each $(t_1, t_2)$ pair
                    \item Apply filter conditions per window
                    \item Cluster qualifying fits by scale
                    \item Compute scale-specific CI
                \end{enumerate}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Multi-Scale LPPLS: Dashboard}

\begin{frame}{Anti-Bubbles and Negative Bubbles}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Definition {\scriptsize[\href{https://doi.org/10.1080/14697688.2002.0000014}{\textcolor{white}{Sornette \& Zhou, 2002}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Anti-bubble}: super-exponential \textbf{decline} with log-periodic oscillations, ending in a \textbf{rally}
                    \item $\ln P(t) = A + B'(t - t_c')^m[1 + C'\cos(\omega\ln(t - t_c') + \phi')]$
                    \item $t > t_c'$ (post-critical), $B' > 0$ (declining toward $A$)
                \end{itemize}
            \end{block}
            \vspace{-1mm}
            \begin{exampleblock}{Key Difference}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Bubble: $t < t_c$, price accelerates \textbf{up}
                    \item Anti-bubble: $t > t_c'$, price decelerates \textbf{down}
                    \item Both show log-periodic oscillations
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{block}{Historical Examples}
                \begin{itemize}\setlength{\itemsep}{1pt}
                    \item \textbf{Japan 1990--2003:} Nikkei anti-bubble after 1989 peak, lasting $>$13 years
                    \item \textbf{S\&P 500 2000--2003:} Post-dot-com decline with clear log-periodicity
                    \item \textbf{Oil 2008--2009:} \$147 $\to$ \$32, anti-bubble pattern
                \end{itemize}
            \end{block}

            \begin{block}{Practical Implications}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Detect \textbf{capitulation points} (end of bear markets)
                    \item Signal regime transitions to recovery
                    \item Symmetric framework: same code, flipped sign
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Bayesian LPPL Estimation}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Motivation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Classical LPPL uses point estimates; \textbf{Bayesian LPPL} [Brée \& Joseph, 2013; Geraskin \& Fantazzini, 2013] provides:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item Full posterior $p(t_c, m, \omega \mid \text{data})$
                        \item Credible intervals (not bootstrap)
                        \item Model comparison via Bayes factors
                    \end{itemize}
                \end{itemize}
            \end{block}

            \begin{exampleblock}{Prior Specification}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Weakly informative priors:
                    \begin{itemize}\setlength{\itemsep}{0pt}
                        \item $m \sim \text{Uniform}(0.1, 0.9)$
                        \item $\omega \sim \text{Uniform}(4, 25)$
                        \item $t_c \sim \text{Uniform}(t_N, t_N + 0.5T)$
                        \item $\sigma \sim \text{Half-Cauchy}(0, 1)$
                    \end{itemize}
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{MCMC Implementation (HMC)}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Define likelihood: $\varepsilon_t \sim \mathcal{N}(0, \sigma^2)$
                    \item Set priors on $(t_c, m, \omega, A, B, C_1, C_2)$
                    \item Run HMC with 4 chains, 2000 warmup
                    \item Check convergence ($\hat{R} < 1.01$)
                \end{enumerate}
            \end{alertblock}

            \begin{block}{Advantages over Bootstrap}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Incorporates prior knowledge
                    \item Handles multimodality naturally
                    \item Provides model evidence for comparison
                    \item Tools: \texttt{PyMC}, \texttt{Stan}, \texttt{NumPyro}
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Regime-Switching LPPL}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Framework {\scriptsize[\href{https://doi.org/10.1016/j.physa.2020.125367}{\textcolor{white}{Ardila-Alvarez et al., 2021}}]}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL + \textbf{Markov-Switching}; $S_t \in \{$Normal, Bubble, Post-crash$\}$
                \end{itemize}
                \vspace{-2mm}
                $$\ln P_t = \begin{cases}
                    \mu + \sigma\varepsilon_t & S_t = \text{Normal} \\
                    \text{LPPL}(t) + \sigma_b\varepsilon_t & S_t = \text{Bubble} \\
                    \alpha + \beta t + \sigma_c\varepsilon_t & S_t = \text{Post}
                \end{cases}$$
            \end{block}
            \vspace{-2mm}
            \begin{exampleblock}{Transition Matrix}
                \vspace{-2mm}
                $$P = \begin{pmatrix}
                    p_{NN} & p_{NB} & 0 \\
                    0 & p_{BB} & p_{BP} \\
                    p_{PN} & 0 & p_{PP}
                \end{pmatrix}$$
                \vspace{-4mm}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item No direct Normal $\to$ Post-crash transition
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Key Innovation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item \textbf{Endogenous regime detection:} No need to pre-specify bubble windows
                    \item \textbf{Filtered probabilities:} $P(S_t = \text{Bubble} \mid \mathcal{F}_t)$ in real time
                    \item \textbf{Duration modeling:} Expected time in bubble state
                \end{itemize}
            \end{exampleblock}
            \vspace{-2mm}
            \begin{block}{Estimation}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item EM algorithm for regime parameters
                    \item Nested LPPL optimization in bubble state
                    \item Hamilton filter for state probabilities
                    \item AIC/BIC for number of regimes
                \end{enumerate}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{Academic Critiques}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Statistical Concerns {\scriptsize[\href{https://doi.org/10.1080/14697688.2011.607467}{\textcolor{white}{Bree et al., 2013}}]}}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item \textbf{Overfitting:} 7 parameters for $\sim$100 points
                    \item \textbf{Look-ahead bias:} easy to fit ex-post
                    \item \textbf{Selection bias:} only successes reported
                    \item \textbf{Multiple testing:} many windows, some will fit
                    \item \textbf{Data snooping:} parameters tuned to known crashes
                \end{enumerate}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{Response to Critiques}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Strict filters (all 8) + out-of-sample evaluation
                    \item Report CIs, false-positive rates, placebo tests
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Empirical Concerns}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item False positives occur
                    \item Timing uncertainty is large
                    \item Not all bubbles show LPPL patterns
                    \item Not all LPPL fits lead to crashes
                    \item Parameters vary across markets
                \end{enumerate}
            \end{alertblock}
            \vspace{-2mm}
            \begin{block}{The Honest Assessment}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPL is a \textbf{fever thermometer}: elevated signal $\Rightarrow$ elevated risk
                    \item Does not guarantee crash; use as one input among many
                    \item Humility is essential
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{When LPPL Signals Fail}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Failure Modes}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Market enters \textbf{sideways correction} instead of crashing
                    \item Estimated $t_c$ \textbf{shifts forward} as new data arrive
                    \item Bubble \textbf{deflates gradually} (no sharp crash)
                    \item Signal appears only for \textbf{narrow windows}
                    \item Log-periodicity may be a \textbf{fitting artefact}
                    \item High $R^2$ does not prove \textbf{predictive validity}
                \end{enumerate}
            \end{alertblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Historical False Positives}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item US equities 2014--2015: LPPL signals, no crash
                    \item Bitcoin 2019: brief LPPL signal, gradual decline
                    \item European debt 2011: exogenous policy, not LPPL bubble
                \end{itemize}
            \end{alertblock}

            \begin{block}{Honest Interpretation}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item A false positive may still indicate overheating or fragility
                    \item But it \textbf{must be reported honestly}
                    \item Never hide failures behind successful case studies
                \end{itemize}
            \end{block}
        \end{column}

\begin{frame}{Ambiguous Signals: When Warnings Do Not Become Crashes}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{How LPPL Warnings Can Be ``Right'' But Not Useful}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item Market enters a \textbf{sideways correction} instead of crashing
                    \item Estimated $t_c$ \textbf{moves forward} as new data arrive (``rolling $t_c$'')
                    \item Bubble \textbf{deflates gradually} rather than bursting (slow air leak)
                    \item Only a \textbf{narrow subset} of windows passes the filters
                    \item Apparent oscillations \textbf{disappear} under placebo testing
                    \item Signal is valid as \textbf{fragility evidence} but invalid as a crash-timing signal
                \end{enumerate}
            \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{Example: S\&P 500, 2014--2015}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item LPPLS CI rose above 0.5 in late 2014
                    \item Market continued rising for 6 months
                    \item August 2015: $-12\%$ correction (not a crash)
                    \item LPPL signal was consistent with overheating but timing was off by months
                \end{itemize}
            \end{alertblock}
            \vspace{-2mm}
            \begin{exampleblock}{Key Message}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item A false positive is not necessarily useless: it may indicate overheating or fragility
                    \item But it must be \textbf{reported honestly} and separated from successful warnings
                    \item The scientific question is not ``Was there a crash?'' but ``Was the regime consistent with endogenous instability?''
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
    }
    \end{cminipage}
\end{frame}

\begin{frame}{What LPPL Can and Cannot Do}
    \begin{cminipage}{0.95\textwidth}
    {\footnotesize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{LPPL \textbf{Can}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Detect super-exponential price acceleration
                    \item Estimate the most likely end-of-bubble regime
                    \item Provide probabilistic risk windows
                    \item Quantify log-periodic oscillations
                    \item Serve as one input in a multi-indicator framework
                    \item Identify non-bubble patterns (negative control)
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{alertblock}{LPPL \textbf{Cannot}}
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item Determine the exact date or magnitude of a crash
                    \item Guarantee that a regime break will occur
                    \item Detect exogenous shocks (pandemics, wars)
                    \item Work reliably on short windows ($< 100$ days)
                    \item Replace fundamental analysis or due diligence
                    \item Distinguish bubbles from genuine regime shifts
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
    \vspace{-1mm}
    \begin{block}{The Bottom Line}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item LPPL is a \textbf{diagnostic tool}, not a crystal ball
            \item Identifies a statistical pattern consistent with endogenous bubble dynamics
            \item A positive signal indicates elevated instability; it does not make a crash certain
        \end{itemize}

\section{Conclusion}

\begin{frame}{Summary: Key Takeaways}
    \begin{cminipage}{0.95\textwidth}
    {\scriptsize
    \begin{columns}[T]
        \begin{column}{0.47\textwidth}
            \begin{block}{Theoretical Foundation}
                \begin{enumerate}\setlength{\itemsep}{0pt}
                    \item Bubbles show \textbf{super-exponential} growth
                    \item LPPL interprets bubbles as \textbf{herding} + positive feedback
                    \item Exhibit \textbf{log-periodic} oscillations
                    \item Same physics as \textbf{phase transitions}
                \end{enumerate}
            \end{block}
            \vspace{-3mm}
            \begin{exampleblock}{The LPPL Model}
                $$\ln P = A + B(t_c-t)^m + C(t_c-t)^m\cos(\omega\ln(t_c-t)-\phi)$$
                \begin{itemize}\setlength{\itemsep}{0pt}
                    \item 7 parameters (3 linear, 4 nonlinear)
                    \item Partial linearization + DE estimation
                    \item 8 filter conditions for validity
                    \item \textbf{Bootstrap CI} for uncertainty
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.47\textwidth}
            \begin{exampleblock}{Case Study Results (Mean error: 10.2d)}
                {\scriptsize
                \vspace{-1mm}
                \begin{center}
                    \begin{tabular}{lcc}

                        Shanghai 2015 & 27d & [169, 187] \\
                        Oil 2008 & 12d & [382, 423] \\
                        COVID 2020 & N/A & No bubble \\
                        \bottomrule
                    \end{tabular}
                \end{center}
                }
            \end{exampleblock}
            \vspace{-2mm}
            \begin{block}{Key Message}
                \begin{itemize}\setlength{\itemsep}{0pt}

            \item Used responsibly $\Rightarrow$ early-warning systems, stress testing, risk review
