\begin{frame}{Exercise 2: The LPPL Equation}
    \begin{cminipage}{0.95\textwidth}
    \begin{alertblock}{Question}
        The LPPL model describes log-price during a bubble:
        $\ln P(t) = A + B(t_c - t)^m + C(t_c - t)^m \cos\!\bigl(\omega \ln(t_c - t) - \varphi\bigr)$
    \end{alertblock}

    \vspace{0.1cm}

    \begin{block}{Exercise}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textcolor{MainBlue}{\textbf{(A)}} Using synthetic parameters ($A{=}11$, $B{=}{-}0.5$, $m{=}0.5$, $C{=}0.04$, $\omega{=}8$, $\varphi{=}1$, $t_c{=}300$), plot the three components: power-law trend, log-periodic oscillations, full LPPL.
            \item \textcolor{MainBlue}{\textbf{(B)}} Why must $B < 0$ and $0 < m < 1$ for a bubble? What happens if $m \geq 1$?
            \item \textcolor{MainBlue}{\textbf{(C)}} Why do oscillations accelerate in calendar time but have constant frequency in $\ln(t_c - t)$?
        \end{itemize}
    \end{block}

    \vspace{0.1cm}

    \begin{center}
        \textit{Answer on next slide...}
    \end{center}
    \end{cminipage}
\end{frame}

\begin{frame}{Exercise 2: Solution}
    \begin{cminipage}{0.95\textwidth}
    \begin{exampleblock}{Solution}
        {\small
        \begin{itemize}\setlength{\itemsep}{3pt}
            \item \textcolor{MainBlue}{\textbf{(A)}} Component 1: $A + B(t_c - t)^m$ curves upward as $t \to t_c$ (power-law singularity). Component 2: $C(t_c - t)^m \cos(\omega \ln(t_c - t) - \varphi)$ --- oscillations that shrink in amplitude and compress in period near $t_c$. Combined: super-exponential trend decorated with accelerating oscillations.
            \item \textcolor{MainBlue}{\textbf{(B)}} $B < 0$ ensures the function \emph{increases} as $t \to t_c$ (since $(t_c - t)^m$ decreases). $0 < m < 1$ ensures \textbf{super-exponential growth}: the growth rate $|\dot{P}/P|$ diverges as $t \to t_c$ (finite-time singularity). If $m \geq 1$, growth decelerates --- no bubble signature.
            \item \textcolor{MainBlue}{\textbf{(C)}} The oscillation argument is $\omega \ln(t_c - t)$. Equal increments of $\ln(t_c - t)$ correspond to geometric compression in $(t_c - t)$: successive cycles take $1/\lambda$ the time of the previous one, where $\lambda = e^{2\pi/\omega} \approx 2$. This is \textbf{discrete scale invariance}.
        \end{itemize}
        }
    \end{exampleblock}
    \end{cminipage}
    \quantlet{TSA\_ch13\_lppl\_components}{https://github.com/danpele/TSA/tree/main/TSA_ch13/TSA_ch13_lppl_components}
\end{frame}

            \item \textcolor{MainBlue}{\textbf{(C)}} Plot the LPPL fit against observed log-price and the residuals. Are residuals stationary?
        \end{itemize}
    \end{block}

    \vspace{0.15cm}

    \begin{center}

    \end{center}
    \end{cminipage}
\end{frame}

\begin{frame}{Exercise 3: Solution}
    \begin{cminipage}{0.95\textwidth}
    \begin{exampleblock}{Solution}
        {\footnotesize
        \begin{itemize}\setlength{\itemsep}{3pt}
            \item \textcolor{MainBlue}{\textbf{(A)}} For fixed $(t_c, m, \omega)$, define $\Delta_i = t_c - t_i$ and $f_i = \Delta_i^m$:
            \[
                \mathbf{X} = \begin{bmatrix} 1 & f_1 & f_1\cos(\omega\ln\Delta_1) & f_1\sin(\omega\ln\Delta_1) \\ \vdots & \vdots & \vdots & \vdots \\ 1 & f_n & f_n\cos(\omega\ln\Delta_n) & f_n\sin(\omega\ln\Delta_n) \end{bmatrix}, \quad \boldsymbol{\beta} = \begin{pmatrix} A \\ B \\ C_1 \\ C_2 \end{pmatrix}
            \]
            \item \textcolor{MainBlue}{\textbf{(B)}} Bounds: $t_c \in [N{-}5, N{+}60]$, $m \in [0.1, 0.9]$, $\omega \in [4, 25]$. Typical BTC 2021 results: $t_c \approx N + 5$, $m \approx 0.4$, $\omega \approx 8$, $R^2 > 0.99$.
            \item \textcolor{MainBlue}{\textbf{(C)}} The fit closely tracks the observed log-price. Residuals fluctuate around zero with small magnitude; check stationarity via Ljung-Box test on residuals.
        \end{itemize}
        }
    \end{exampleblock}

\begin{frame}{Exercise 4: The Eight Filter Conditions}
    \begin{cminipage}{0.95\textwidth}
    \begin{alertblock}{Question}

            \item \textcolor{MainBlue}{\textbf{(B)}} Check all 8 conditions on your BTC 2021 LPPL fit. Display pass/fail for each.
            \item \textcolor{MainBlue}{\textbf{(C)}} Why is the condition $B < 0$ essential? What would $B > 0$ mean economically?
        \end{itemize}
    \end{block}

    \vspace{0.15cm}

    \begin{center}
        \textit{Answer on next slide...}
    \end{center}
    \end{cminipage}
\end{frame}

\begin{frame}{Exercise 4: Solution}
    \begin{cminipage}{0.95\textwidth}
    \begin{exampleblock}{Solution}
        {\scriptsize
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textcolor{MainBlue}{\textbf{(A)}} The 8 conditions:
            \begin{enumerate}\setlength{\itemsep}{0pt}
                \item $B < 0$: super-exponential growth (price rises as $t \to t_c$)
                \item $0.1 \leq m \leq 0.9$: power-law exponent in empirical range
                \item $4 \leq \omega \leq 25$: log-frequency in observed range
                \item $t_c > N$: critical time after the last observation
                \item $t_c < N + 0.2N$: critical time within a reasonable horizon
                \item $|C| > 0$: log-periodic oscillations are present
                \item $m|B| > 0$: finite-time singularity has detectable strength
                \item $\omega / (2\pi) \geq 1$: at least one full oscillation cycle
            \end{enumerate}
            \item \textcolor{MainBlue}{\textbf{(B)}} For BTC 2021, typically 8/8 conditions pass.
            \item \textcolor{MainBlue}{\textbf{(C)}} $B > 0$ would mean price \emph{decreases} as $t \to t_c$ --- this describes an \textbf{anti-bubble} (deflationary spiral), not a speculative bubble.
        \end{itemize}
        }
    \end{exampleblock}
    \end{cminipage}

            \item \textcolor{MainBlue}{\textbf{(B)}} Plot histograms of $t_c$, $m$, and $\omega$ from the bootstrap. Report 95\% CIs. Convert the $t_c$ CI to calendar dates.
            \item \textcolor{MainBlue}{\textbf{(C)}} Create a scatter plot of $m$ vs.\ $\omega$ from the bootstrap. What does the shape of the joint distribution tell us about parameter identifiability?
        \end{itemize}
    \end{block}

    \vspace{0.1cm}

    \begin{center}
        \textit{Answer on next slide...}
    \end{center}
    \end{cminipage}
\end{frame}

\begin{frame}{Exercise 6: Solution}
    \begin{cminipage}{0.95\textwidth}
    \begin{exampleblock}{Solution}
        {\small
        \begin{itemize}\setlength{\itemsep}{3pt}
            \item \textcolor{MainBlue}{\textbf{(A)}} The residual bootstrap preserves the LPPL structure: $y^*_t = \hat{y}_t + \varepsilon^*_t$ where $\varepsilon^*_t$ are drawn with replacement from the original residuals $\hat{\varepsilon}_t = y_t - \hat{y}_t$. Each bootstrap sample is re-fitted from scratch using differential evolution.
            \item \textcolor{MainBlue}{\textbf{(B)}} For BTC 2021, a typical 95\% CI for $t_c$: approximately $\pm 5$--10 trading days around the point estimate. $m$: typically tight CI $\in [0.3, 0.6]$. $\omega$: moderate spread reflecting the challenge of pinpointing oscillation frequency.
            \item \textcolor{MainBlue}{\textbf{(C)}} The $m$--$\omega$ scatter often shows a negative correlation: higher $m$ values pair with lower $\omega$. A tight ellipse indicates good identifiability; a diffuse cloud suggests the parameters are poorly identified (multiple local optima).
        \end{itemize}
        }
    \end{exampleblock}
    \end{cminipage}

        \begin{itemize}\setlength{\itemsep}{3pt}
            \item \textcolor{MainBlue}{\textbf{(A)}} The position shrinks smoothly from 100\% to 10\% as CI rises. This avoids binary ``all-in / all-out'' decisions and provides proportional exposure reduction.
            \item \textcolor{MainBlue}{\textbf{(B)}} The LPPL-adjusted strategy typically achieves: lower maximum drawdown, lower daily volatility, and higher risk-adjusted returns (Sharpe ratio) compared to buy-and-hold, at the cost of slightly lower total return.
            \item \textcolor{MainBlue}{\textbf{(C)}} Limitations: (i) parameter estimation delay --- each LPPL fit requires $>60$ days of data, so the CI lags real-time price action; (ii) transaction costs from position changes can erode returns; (iii) look-ahead bias if filter conditions reference the in-sample $t_c$; (iv) false positives --- some windows may produce spurious bubble signals; (v) LPPL cannot predict the exact crash date, only the approximate critical time window.
        \end{itemize}
        }
    \end{exampleblock}
    \end{cminipage}

\end{frame}

    \begin{block}{Topic 1: EMH vs.\ LPPL}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item EMH says crashes are unpredictable random events. LPPL says some crashes are preceded by detectable super-exponential growth. Can both be partially correct?

    \begin{alertblock}{Topic 2: Overfitting Risk}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item LPPL has 7 parameters fitted to sometimes $<100$ observations. How do filter conditions, AIC/BIC, and bootstrap protect against overfitting?
            \item Would you trust an LPPL signal for a real trading decision?
        \end{itemize}
    \end{alertblock}

    \vspace{-2mm}

    \begin{exampleblock}{Topic 3: Ethical Dimensions}
        \begin{itemize}\setlength{\itemsep}{0pt}
            \item If a central bank detects a bubble using LPPL, should it act preemptively?
            \item If a hedge fund uses LPPL to time a market exit, could this become self-defeating if widely adopted?
        \end{itemize}
    \end{exampleblock}
